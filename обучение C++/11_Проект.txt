## Цели урока
- Объединить все изученные темы в один проект
- Практика проектирования классов и функций

---

## Требования к проекту

Создать консольное приложение **Task Manager** с функционалом:
1. Добавление задач (название, описание, приоритет, дедлайн, теги)
2. Просмотр списка задач (фильтры: все, активные, выполненные, по приоритету)
3. Отметка задачи как выполненной / удаление
4. Сохранение/загрузка из файла (JSON или простой текстовый формат)
5. Поиск по ключевым словам
6. Статистика: всего задач, выполнено, просрочено

#pragma once
#include "task.h"
#include <vector>
#include <optional>

class TaskManager {
    std::vector<Task> tasks_;
    int next_id_ = 1;

    // CRUD

                   const std::vector<std::string>& tags);
    bool remove_task(int id);

    // Запросы

    std::vector<Task*> get_by_priority(Priority) const;
    std::vector<Task*> search(const std::string& keyword) const;
    std::vector<Task*> get_overdue() const;

    // Статистика

## Реализация (ключевые фрагменты)

### src/task.cpp
```cpp
#include "task.h"
#include <iomanip>
#include <sstream>

bool Task::is_overdue() const {
    if (status == Status::Completed) return false;
    if (!due_date) return false;
    return *due_date < std::chrono::system_clock::now();
}

std::string Task::priority_str() const {
    switch (priority) {
        case Priority::Low: return "Low";
        case Priority::Medium: return "Medium";
        case Priority::High: return "High";
        case Priority::Critical: return "Critical";
    }
    return "?";
}

std::string Task::due_date_str() const {
    if (!due_date) return "—";
    auto t = std::chrono::system_clock::to_time_t(*due_date);
    std::stringstream ss;
    ss << std::put_time(std::localtime(&t), "%Y-%m-%d %H:%M");
    return ss.str();
}

std::ostream& operator<<(std::ostream& os, const Task& t) {
    os << "[" << t.id << "] " << t.title
       << " | " << t.priority_str()
       << " | " << t.status_str()
       << " | Due: " << t.due_date_str();
    if (!t.tags.empty()) {
        os << " | Tags: ";
        for (size_t i = 0; i < t.tags.size(); ++i) {
            if (i) os << ", ";
            os << t.tags[i];
        }
    }
    return os;
}
```

### src/task_manager.cpp (фрагменты)
```cpp
#include "task_manager.h"
#include "storage.h"
#include <algorithm>
#include <fstream>

TaskManager::TaskManager(const std::string& path) : storage_path_(path) {
    load();
}

Task& TaskManager::add_task(const std::string& title, const std::string& desc,
                            Priority p, const std::optional<std::chrono::system_clock::time_point>& due,
                            const std::vector<std::string>& tags) {
    Task t;
    t.id = next_id_++;
    t.title = title;
    t.description = desc;
    t.priority = p;
    t.due_date = due;
    t.tags = tags;
    t.created_at = std::chrono::system_clock::now();
    tasks_.push_back(std::move(t));
    save();
    return tasks_.back();
}

bool TaskManager::remove_task(int id) {
    auto it = std::find_if(tasks_.begin(), tasks_.end(),
                           [id](const Task& t){ return t.id == id; });
    if (it == tasks_.end()) return false;
    tasks_.erase(it);
    save();
    return true;
}

std::optional<Task*> TaskManager::find_task(int id) {
    for (auto& t : tasks_) if (t.id == id) return &t;
    return std::nullopt;
}

TaskManager::Stats TaskManager::get_stats() const {
    Stats s{};
    auto now = std::chrono::system_clock::now();
    for (const auto& t : tasks_) {
        ++s.total;
        if (t.status == Status::Completed) ++s.completed;
        else {
            ++s.pending;
            if (t.due_date && *t.due_date < now) ++s.overdue;
        }
    }
    return s;
}

bool TaskManager::save() {
    JsonStorage storage;
    return storage.save(tasks_, storage_path_);
}
bool TaskManager::load() {
    JsonStorage storage;
    return storage.load(tasks_, storage_path_);
}
```

## Главное меню (main.cpp)

```cpp
#include "task_manager.h"
#include "utils.h"
#include <iostream>
#include <string>

void print_menu() {
    std::cout << R"(
=== Task Manager ===
1. Add task
2. List all tasks
3. List pending
4. List completed
5. List overdue
6. Search tasks
7. Complete/Uncomplete task
8. Delete task
9. Show statistics
0. Exit
Choose: )";
}

int main() {
    TaskManager tm("data/tasks.json");
    int choice = -1;

    while (choice != 0) {
        print_menu();
        if (!(std::cin >> choice)) { std::cin.clear(); std::cin.ignore(10000,'\n'); continue; }
        std::cin.ignore(10000, '\n');

        switch (choice) {
            case 1: {
                std::string title = input_line("Title: ");
                std::string desc = input_line("Description (optional): ");
                Priority p = input_priority("Priority (low/medium/high/critical): ");
                auto due = input_date("Due date (YYYY-MM-DD HH:MM, empty=none): ");
                auto tags = input_tags("Tags (comma separated): ");
                tm.add_task(title, desc, p, due, tags);
                std::cout << "Task added!\n";
                break;
            }
            case 2: case 3: case 4: case 5: {
                std::vector<Task*> list;
                if (choice == 2) list = tm.get_all();
                else if (choice == 3) list = tm.get_by_status(Status::Pending);
                else if (choice == 4) list = tm.get_by_status(Status::Completed);
                else list = tm.get_overdue();
                print_tasks(list);
                break;
            }
            case 6: {
                std::string kw = input_line("Keyword: ");
                auto results = tm.search(kw);
                print_tasks(results);
                break;
            }
            case 7: {
                int id = input_int("Task ID: ");
                if (tm.toggle_status(id)) std::cout << "Toggled!\n";
                else std::cout << "Not found\n";
                break;
            }
            case 8: {
                int id = input_int("Task ID: ");
                if (tm.remove_task(id)) std::cout << "Deleted!\n";
                else std::cout << "Not found\n";
                break;
            }
            case 9: {
                auto s = tm.get_stats();
                std::cout << "Total: " << s.total
                          << ", Done: " << s.completed
                          << ", Pending: " << s.pending
                          << ", Overdue: " << s.overdue << '\n';
                break;
            }
            case 0: std::cout << "Goodbye!\n"; break;
            default: std::cout << "Invalid option\n";
        }
    }
}
```
