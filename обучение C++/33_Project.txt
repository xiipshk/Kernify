## 1. Project Specification: High-Performance Log Processing Library

### Requirements
```
Name: fastlog — high-throughput structured logging library
Target: C++20, cross-platform (Linux, Windows, macOS)
Performance: >10M logs/sec single-threaded, lock-free async backend
Features:
  - Structured logging (key-value pairs, not just strings)
  - Multiple sinks: file, stdout, syslog, network, custom
  - Log levels: trace, debug, info, warn, error, critical
  - Filtering by level, module, custom predicates
  - Formatters: JSON, text, binary (protobuf/flatbuffers)
  - Async processing with bounded queue (backpressure)
  - Zero-cost when disabled (compile-time level filtering)
  - Thread-safe, lock-free hot path
  - Header-only core + compiled backend
```

## 2. Core Implementation Highlights

### Zero-Cost Macros
```cpp
// macros.hpp
#define FASTLOG_TRACE(logger, msg, ...) \
    FASTLOG_LOG(logger, fastlog::Level::Trace, msg, ##__VA_ARGS__)

#define FASTLOG_LOG(logger, level, msg, ...) \
    do { \
        if (FASTLOG_LIKELY((logger).should_log(level))) { \
            (logger).log(level, __FILE__, __LINE__, __FUNCTION__, msg, ##__VA_ARGS__); \
        } \
    } while(0)

// Compile-time level filtering
template <fastlog::Level GlobalLevel = fastlog::Level::Info>
class CompileTimeFilter {
public:
    static constexpr bool should_log(fastlog::Level lvl) noexcept {
        return lvl >= GlobalLevel;
    }
};
```

### Lock-Free Ring Buffer (SPSC)
```cpp
// ring_buffer.hpp
template <typename T, size_t Capacity>
class RingBuffer {
    static_assert((Capacity & (Capacity - 1)) == 0, "Power of 2");
    alignas(64) std::atomic<size_t> head_{0};
    alignas(64) std::atomic<size_t> tail_{0};
    T buffer_[Capacity];
public:
    bool try_push(T&& item) noexcept {
        size_t head = head_.load(std::memory_order_relaxed);
        size_t next = (head + 1) & (Capacity - 1);
        if (next == tail_.load(std::memory_order_acquire)) return false;
        buffer_[head] = std::move(item);
        head_.store(next, std::memory_order_release);
        return true;
    }
    bool try_pop(T& item) noexcept {
        size_t tail = tail_.load(std::memory_order_relaxed);
        if (tail == head_.load(std::memory_order_acquire)) return false;
        item = std::move(buffer_[tail]);
        tail_.store((tail + 1) & (Capacity - 1), std::memory_order_release);
        return true;
    }
};
```

### Async Worker with Backpressure
```cpp
// async_worker.hpp
class AsyncWorker {
    RingBuffer<LogRecord, 65536> queue_;
    std::thread worker_;
    std::atomic<bool> running_{true};
    std::unique_ptr<Sink> sink_;
    std::unique_ptr<Formatter> formatter_;
    BackpressurePolicy policy_;

    void run() {
        LogRecord record;
        while (running_.load(std::memory_order_acquire)) {
            if (queue_.try_pop(record)) {
                auto formatted = formatter_->format(record);
                sink_->write(formatted);
            } else {
                std::this_thread::yield(); // или wait/notify
            }
        }
        // Drain remaining
        while (queue_.try_pop(record)) {
            sink_->write(formatter_->format(record));
        }
    }
public:
    bool try_log(LogRecord&& record) {
        if (!queue_.try_push(std::move(record))) {
            return policy_.handle_overflow(*this, std::move(record));
        }
        return true;
    }
};
```

## 3. Testing Strategy

### Unit Tests (Catch2 / GoogleTest)
```cpp
// tests/unit/level_test.cpp
#include <catch2/catch_test_macros.hpp>
#include <fastlog/level.hpp>

TEST_CASE("Level comparison") {
    using fastlog::Level;
    REQUIRE(Level::Trace < Level::Debug);
    REQUIRE(Level::Error > Level::Warn);
    REQUIRE(Level::Critical == Level::Critical);
}

TEST_CASE("Level from string") {
    REQUIRE(fastlog::level_from_string("debug") == Level::Debug);
    REQUIRE(fastlog::level_from_string("INVALID") == Level::Info); // default
}
```

```cpp
// tests/unit/ring_buffer_test.cpp
TEST_CASE("RingBuffer SPSC") {
    RingBuffer<int, 1024> rb;
    REQUIRE(rb.try_push(42));
    int val;
    REQUIRE(rb.try_pop(val));
    REQUIRE(val == 42);
    REQUIRE_FALSE(rb.try_pop(val));
}

TEST_CASE("RingBuffer overflow") {
    RingBuffer<int, 4> rb;
    for (int i = 0; i < 4; ++i) REQUIRE(rb.try_push(i));
    REQUIRE_FALSE(rb.try_push(99)); // full
}
```

## 4. Benchmarking & Performance Regression

### Google Benchmark Suite
```cpp
// bench/log_throughput.cpp
#include <benchmark/benchmark.h>
#include <fastlog/logger.hpp>

static void BM_SyncLogging(benchmark::State& state) {
    auto logger = fastlog::Logger::create("bench")
        .sink(std::make_shared<NullSink>())
        .level(fastlog::Level::Info)
        .build();

    for (auto _ : state) {
        FASTLOG_INFO(logger, "Message {}", "with args");
    }
    state.SetItemsProcessed(state.iterations());
}
BENCHMARK(BM_SyncLogging);

static void BM_AsyncLogging(benchmark::State& state) {
    auto logger = fastlog::Logger::create("bench")
        .async(65536)
        .sink(std::make_shared<NullSink>())
        .level(fastlog::Level::Info)
        .build();

    for (auto _ : state) {
        FASTLOG_INFO(logger, "Message {}", "with args");
    }
    state.SetItemsProcessed(state.iterations());
}
BENCHMARK(BM_AsyncLogging);

static void BM_DisabledLogging(benchmark::State& state) {
    auto logger = fastlog::Logger::create("bench")
        .level(fastlog::Level::Warn)  // Info disabled
        .build();

    for (auto _ : state) {
        FASTLOG_INFO(logger, "This should be no-op");
    }
    state.SetItemsProcessed(state.iterations());
}
BENCHMARK(BM_DisabledLogging);

BENCHMARK_MAIN();
```

```cpp
// include/fastlog/logger.hpp
/**
 * @file logger.hpp
 * @brief Main logger interface

namespace fastlog {

/**
 * @class Logger
 * @brief Thread-safe structured logger with async support
 *
 * Example usage:
 * @code
 * auto logger = Logger::create("myapp")
 *     .async(65536)
 *     .sink(std::make_shared<FileSink>("app.log"))
 *     .level(Level::Info)
 *     .build();

 * FASTLOG_INFO(logger, "User logged in", "user_id", 42);

class Logger {
public:
    /**
     * @brief Create logger builder
     * @param name Logger name (used in output)
     * @return LoggerBuilder for fluent configuration
     */
    static LoggerBuilder create(std::string_view name);

    /**
     * @brief Log a message at specified level
     * @tparam Args Key-value pairs (key, value, key, value...)
     * @param level Log level
     * @param file Source file (__FILE__)
     * @param line Source line (__LINE__)
     * @param func Source function (__FUNCTION__)
     * @param message Format string
     * @param args Key-value arguments
     */
    template <typename... Args>
    void log(Level level, const char* file, int line, const char* func,
             std::string_view message, Args&&... args);
};

} // namespace fastlog
```

```yaml
# .github/workflows/ci.yml

jobs:
  build-test:
    strategy:
      matrix:
        os: [ubuntu-latest, windows-latest, macos-latest]
        compiler: [gcc-12, gcc-13, clang-15, clang-16, msvc]
        include:
          - os: ubuntu-latest
            compiler: gcc-12
          - os: ubuntu-latest
            compiler: clang-15
          - os: windows-latest
            compiler: msvc
          - os: macos-latest
            compiler: clang-16
    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4
      - name: Install dependencies
        run: |
          # vcpkg / conan / apt / brew
      - name: Configure
        run: cmake -B build -DCMAKE_BUILD_TYPE=Release -DENABLE_SANITIZERS=ON -DBUILD_TESTS=ON
      - name: Build
        run: cmake --build build --parallel
      - name: Unit tests
        run: ctest --test-dir build --output-on-failure -j4
      - name: Integration tests
        run: ctest --test-dir build -L integration --output-on-failure
      - name: Fuzzing (nightly only)
        if: github.event_name == 'schedule'
        run: |
          cmake -B build-fuzz -DSANITIZE_FUZZ=ON
          cmake --build build-fuzz --target fuzz_formatter
          ./build-fuzz/tests/fuzz/fuzz_formatter -max_total_time=300

    runs-on: ubuntu-latest

      - uses: actions/checkout@v4
      - name: Build docs

        with:
          github_token: ${{ secrets.GITHUB_TOKEN }}

      - uses: actions/checkout@v4

      - name: Publish to vcpkg/Conan
        run: |
          # vcpkg port submission / conan upload
```

### Conan v2 Recipe
```python
# conanfile.py
from conan import ConanFile
from conan.tools.cmake import CMake, cmake_layout

class FastlogConan(ConanFile):
    name = "fastlog"
    version = "1.0.0"
    license = "MIT"
    url = "https://github.com/username/fastlog"
    settings = "os", "compiler", "build_type", "arch"
    options = {
        "shared": [True, False],
        "fPIC": [True, False],
        "async": [True, False],
    }
    default_options = {
        "shared": False,
        "fPIC": True,
        "async": True,
    }
    exports_sources = "src/*", "include/*", "CMakeLists.txt", "cmake/*"

    def layout(self):
        cmake_layout(self)

    def requirements(self):
        self.requires("fmt/10.2.1")
        if self.options.async:
            self.requires("concurrentqueue/1.0.3")

    def build(self):
        cmake = CMake(self)
        cmake.configure(variables={"BUILD_TESTS": False})
        cmake.build()

    def package(self):
        cmake = CMake(self)
        cmake.install()

    def package_info(self):
        self.cpp_info.libs = ["fastlog"]
        if self.settings.os == "Linux":
            self.cpp_info.system_libs = ["pthread", "dl"]
```
