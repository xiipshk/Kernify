## 1. Потоки (`std::thread`)

### Создание потоков
```rust
use std::thread;
use std::time::Duration;

fn main() {
    // Простой поток
    let handle = thread::spawn(|| {
        for i in 1..=5 {
            println!("Thread: {}", i);
            thread::sleep(Duration::from_millis(100));
        }
    });

    // Основной поток
    for i in 1..=3 {
        println!("Main: {}", i);
        thread::sleep(Duration::from_millis(100));
    }

    // Ждём завершения
    handle.join().unwrap();
}
```

### Передача данных в поток (move)
```rust
let data = vec![1, 2, 3];

let handle = thread::spawn(move || {
    println!("Data in thread: {:?}", data);
    data // Можно вернуть владение
});

let returned = handle.join().unwrap();
println!("Returned: {:?}", returned);
```

### Scoped threads (все потоки завершаются перед выходом из scope)
```rust
use std::thread;

fn main() {
    let mut data = vec![1, 2, 3];

    thread::scope(|s| {
        s.spawn(|| {
            println!("Read: {:?}", data);
        });
        s.spawn(|| {
            // Можно мутировать, если нет других доступов
            data.push(4);
        });
    });

    // Гарантированно все потоки завершены здесь
    println!("Final: {:?}", data);
}
```

## 2. `Send` и `Sync` — маркеры потокобезопасности

### Определения
```rust
// Send: можно передать владение (move) в другой поток
// Почти все типы: Send, кроме Rc, RefCell, *const T, *mut T

// Sync: безопасно делить &T между потоками
// T: Sync <=> &T: Send
// Почти все типы: Sync, кроме Cell, RefCell, MutexGuard (в некоторых случаях)
```

### Проверка
```rust
fn assert_send<T: Send>() {}
fn assert_sync<T: Sync>() {}

assert_send::<i32>();        // OK
assert_send::<String>();     // OK
assert_send::<Vec<i32>>();   // OK
// assert_send::<Rc<i32>>();  // Error: Rc не Send

assert_sync::<i32>();        // OK
assert_sync::<Mutex<i32>>(); // OK
// assert_sync::<RefCell<i32>>(); // Error: RefCell не Sync
```

### Реализация вручную (unsafe)
```rust
use std::marker::PhantomData;

struct MyPtr<T> {
    ptr: *mut T,
    _marker: PhantomData<T>, // Для dropck
}

unsafe impl<T: Send> Send for MyPtr<T> {}
unsafe impl<T: Sync> Sync for MyPtr<T> {}
```

## 3. Разделяемое состояние

### `Arc<Mutex<T>>` — классический паттерн
```rust
use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    let counter = Arc::new(Mutex::new(0));
    let mut handles = vec![];

    for _ in 0..10 {
        let counter = Arc::clone(&counter);
        handles.push(thread::spawn(move || {
            let mut num = counter.lock().unwrap();
            *num += 1;
        }));
    }

    for h in handles { h.join().unwrap(); }
    println!("Result: {}", *counter.lock().unwrap()); // 10
}
```

### `Arc<RwLock<T>>` — много читателей
```rust
use std::sync::{Arc, RwLock};

fn main() {
    let data = Arc::new(RwLock::new(vec![1, 2, 3]));
    let mut handles = vec![];

    // Читатели
    for _ in 0..5 {
        let data = Arc::clone(&data);
        handles.push(thread::spawn(move || {
            let read = data.read().unwrap();
            println!("Read: {:?}", *read);
        }));
    }

    // Писатель
    let data = Arc::clone(&data);
    handles.push(thread::spawn(move || {
        let mut write = data.write().unwrap();
        write.push(4);
    }));

    for h in handles { h.join().unwrap(); }
}
```

## 4. Каналы (`std::sync::mpsc`)

### Базовый канал
```rust
use std::sync::mpsc;
use std::thread;

fn main() {
    let (tx, rx) = mpsc::channel();

    thread::spawn(move || {
        let vals = vec![1, 2, 3];
        for v in vals {
            tx.send(v).unwrap();
        }
    });

    // rx — итератор, блокируется до получения
    for received in rx {
        println!("Got: {}", received);
    }
}
```

### Множественные отправители
```rust
let (tx, rx) = mpsc::channel();

for i in 0..3 {
    let tx = tx.clone();
    thread::spawn(move || {
        tx.send(i).unwrap();
    });
}

drop(tx); // Важно! Иначе rx никогда не закончится
for v in rx { println!("{}", v); }
```

### `sync_channel` — с буфером (backpressure)
```rust
let (tx, rx) = mpsc::sync_channel(2); // Буфер 2

tx.send(1).unwrap(); // Сразу
tx.send(2).unwrap(); // Сразу
// tx.send(3).unwrap(); // Блокирует, пока не освободится место
```

## 5. Атомарные типы (`std::sync::atomic`)

### Базовые атомарные
```rust
use std::sync::atomic::{AtomicBool, AtomicUsize, AtomicU64, Ordering};
use std::thread;

fn main() {
    let flag = AtomicBool::new(false);
    let counter = AtomicUsize::new(0);

    let handles: Vec<_> = (0..10).map(|_| {
        let flag = &flag;
        let counter = &counter;
        thread::spawn(move || {
            if !flag.load(Ordering::Acquire) {
                flag.store(true, Ordering::Release);
                println!("First!");
            }
            counter.fetch_add(1, Ordering::SeqCst);
        })
    }).collect();

    for h in handles { h.join().unwrap(); }
    println!("Counter: {}", counter.load(Ordering::SeqCst));
}
```

### Memory Ordering
```rust
// SeqCst — последовательная согласованность (самое строгое, дефолт)
// Acquire — загрузка: видит все записи перед Release в другом потоке
// Release — запись: все предыдущие записи видны после Acquire в другом потоке
// AcqRel — комбинация для RMW операций
// Relaxed — без гарантий упорядочивания (только атомарность)

// Пример: Spinlock с Acquire/Release
struct Spinlock {
    locked: AtomicBool,
}

impl Spinlock {
    fn new() -> Self {
        Spinlock { locked: AtomicBool::new(false) }
    }

    fn lock(&self) {
        while self.locked.compare_exchange_weak(
            false, true,
            Ordering::Acquire,  // Success: acquire
            Ordering::Relaxed   // Failure: relaxed
        ).is_err() {
            std::hint::spin_loop(); // Подсказка процессору
        }
    }

    fn unlock(&self) {
        self.locked.store(false, Ordering::Release);
    }
}
```

### Атомарные операции
```rust
let atomic = AtomicUsize::new(10);

atomic.load(Ordering::SeqCst);           // Чтение
atomic.store(20, Ordering::SeqCst);      // Запись
atomic.swap(30, Ordering::SeqCst);       // Обмен
atomic.fetch_add(5, Ordering::SeqCst);   // +5, вернуть старое
atomic.fetch_sub(3, Ordering::SeqCst);   // -3
atomic.fetch_and(0b111, Ordering::SeqCst);
atomic.fetch_or(0b100, Ordering::SeqCst);
atomic.fetch_xor(0b010, Ordering::SeqCst);

// Compare-and-swap
atomic.compare_exchange(
    15, 20,           // expected, new
    Ordering::SeqCst, Ordering::SeqCst  // success, failure
);
atomic.compare_exchange_weak(...);       // Слабая версия (может spuriously fail)
```

## 6. `Rayon` — параллельные итераторы

### Установка
```toml
[dependencies]
rayon = "1.10"
```

### Базовое использование
```rust
use rayon::prelude::*;

fn main() {
    let data: Vec<i32> = (0..1_000_000).collect();

    // Последовательно
    let sum: i32 = data.iter().sum();

    // Параллельно
    let sum: i32 = data.par_iter().sum();

    // Map/Filter
    let squares: Vec<i32> = data.par_iter()
        .map(|x| x * x)
        .filter(|x| *x % 2 == 0)
        .collect();

    // for_each
    data.par_iter().for_each(|x| {
        // параллельно
    });
}
```

### `par_sort`, `par_chunks`
```rust
let mut v = vec![5, 3, 1, 4, 2];
v.par_sort(); // Параллельная сортировка
v.par_sort_unstable();

// Chunks
for chunk in v.par_chunks_mut(2) {
    chunk.sort();
}
```

### `join` — рекурсивное разделение
```rust
use rayon::join;

fn quicksort(v: &mut [i32]) {
    if v.len() <= 1 { return; }
    let mid = partition(v);
    let (left, right) = v.split_at_mut(mid);
    join(|| quicksort(left), || quicksort(right));
}
```

fn main() {
    let mut data = vec![1, 2, 3];

    scope(|s| {
        s.spawn(|_| {
            println!("Read: {:?}", data);
        });

fn main() {

## 8. Lock-free программирование

### `Arc<AtomicPtr<T>>` — простой lock-free stack
```rust
use std::sync::atomic::{AtomicPtr, Ordering};
use std::ptr;

struct Node<T> {
    value: T,
    next: AtomicPtr<Node<T>>,
}

struct LockFreeStack<T> {
    head: AtomicPtr<Node<T>>,
}

impl<T> LockFreeStack<T> {
    fn new() -> Self {
        LockFreeStack { head: AtomicPtr::new(ptr::null_mut()) }
    }

    fn push(&self, value: T) {
        let node = Box::into_raw(Box::new(Node {
            value,
            next: AtomicPtr::new(ptr::null_mut()),
        }));
        loop {
            let head = self.head.load(Ordering::Acquire);
            unsafe { (*node).next.store(head, Ordering::Relaxed); }
            if self.head.compare_exchange_weak(
                head, node,
                Ordering::Release,
                Ordering::Acquire
            ).is_ok() {
                break;
            }
        }
    }

    fn pop(&self) -> Option<T> {
        loop {
            let head = self.head.load(Ordering::Acquire);
            if head.is_null() { return None; }
            let next = unsafe { (*head).next.load(Ordering::Acquire) };
            if self.head.compare_exchange_weak(
                head, next,
                Ordering::Release,
                Ordering::Acquire
            ).is_ok() {
                let node = unsafe { Box::from_raw(head) };
                return Some(node.value);
            }
        }
    }
}
```

## 9. Избегание Deadlock

### Порядок захвата блокировок
```rust
// ПЛОХО: deadlock возможен
fn transfer_bad(from: &Mutex<i32>, to: &Mutex<i32>, amount: i32) {
    let mut f = from.lock().unwrap();
    let mut t = to.lock().unwrap();
    *f -= amount;
    *t += amount;
}

// ХОРОШО: всегда в одном порядке
fn transfer_good(a: &Mutex<i32>, b: &Mutex<i32>, amount: i32) {
    let (first, second) = if (a as *const _) < (b as *const _) {
        (a, b)
    } else {
        (b, a)
    };
    let mut f = first.lock().unwrap();
    let mut s = second.lock().unwrap();
    *f -= amount;
    *s += amount;
}
```

### `try_lock` — неблокирующая попытка
```rust
use std::sync::Mutex;

let mutex = Mutex::new(0);

if let Ok(mut guard) = mutex.try_lock() {
    *guard += 1;
} else {
    println!("Could not acquire lock");
}
```

## 10. Практические упражнения

### Упражнение 1: Параллельный MapReduce
```rust
use rayon::prelude::*;
use std::collections::HashMap;

fn word_count_parallel(texts: &[String]) -> HashMap<String, usize> {
    texts.par_iter()
        .flat_map(|s| s.split_whitespace())
        .fold(HashMap::new, |mut map, word| {
            *map.entry(word.to_lowercase()).or_insert(0) += 1;
            map
        })
        .reduce(HashMap::new, |mut a, b| {
            for (k, v) in b { *a.entry(k).or_insert(0) += v; }
            a
        })
}
```

### Упражнение 2: Thread pool с работниками
```rust
use std::sync::{Arc, Mutex, mpsc};
use std::thread;

struct ThreadPool {
    workers: Vec<Worker>,
    sender: mpsc::Sender<Job>,
}

type Job = Box<dyn FnOnce() + Send + 'static>;

impl ThreadPool {
    fn new(size: usize) -> Self {
        let (sender, receiver) = mpsc::channel();
        let receiver = Arc::new(Mutex::new(receiver));
        let mut workers = Vec::with_capacity(size);

        for id in 0..size {
            workers.push(Worker::new(id, Arc::clone(&receiver)));
        }

        ThreadPool { workers, sender }
    }

    fn execute<F>(&self, f: F) where F: FnOnce() + Send + 'static {
        self.sender.send(Box::new(f)).unwrap();
    }
}

struct Worker {
    id: usize,
    thread: Option<thread::JoinHandle<()>>,
}

impl Worker {
    fn new(id: usize, receiver: Arc<Mutex<mpsc::Receiver<Job>>>) -> Self {
        let thread = thread::spawn(move || loop {
            let job = receiver.lock().unwrap().recv().unwrap();
            println!("Worker {} got job", id);
            job();
        });
        Worker { id, thread: Some(thread) }
    }
}

impl Drop for ThreadPool {
    fn drop(&mut self) {
        for worker in &mut self.workers {
            if let Some(t) = worker.thread.take() {
                t.join().unwrap();
            }
        }
    }
}
```

### Упражнение 3: Lock-free счетчик с пакетной обработкой
```rust
use std::sync::atomic::{AtomicUsize, Ordering};
use std::thread;

struct BatchedCounter {
    local: AtomicUsize,
    global: AtomicUsize,
    batch_size: usize,
}

impl BatchedCounter {
    fn new(batch_size: usize) -> Self {
        BatchedCounter {
            local: AtomicUsize::new(0),
            global: AtomicUsize::new(0),
            batch_size,
        }
    }

    fn increment(&self) {
        let local = self.local.fetch_add(1, Ordering::Relaxed) + 1;
        if local % self.batch_size == 0 {
            self.global.fetch_add(self.batch_size, Ordering::Relaxed);
            self.local.fetch_sub(self.batch_size, Ordering::Relaxed);
        }
    }

    fn get(&self) -> usize {
        self.global.load(Ordering::Relaxed) + self.local.load(Ordering::Relaxed)
    }
}
```

## 11. Типичные ошибки

| Ошибка | Причина | Решение |
|--------|---------|---------|
| `cannot move out of captured variable` | Захват по ссылке в `move` замыкании | Используйте `move` или клонируйте `Arc` |
| `deadlock` | Два потока ждут друг друга | Всегда берите блокировки в одном порядке |
| `data race` | Несинхронизированный доступ | Используйте `Mutex`/`RwLock`/`Atomic` |
| `channel closed` | Все `Sender` дропнуты | Держите `Sender` живым до конца |
| `thread panicked` | Паника в потоке | `handle.join()` возвращает `Result`, обрабатывайте |
