## 1. Принципы архитектуры в Rust

### Разделение ответственности (Separation of Concerns)
```
src/
├── domain/           # Бизнес-логика, сущности, трейты
│   ├── models.rs     # Domain entities (User, Order, etc.)
│   ├── repositories.rs # Repository traits
│   └── services.rs   # Business logic traits
├── infrastructure/   # Реализации (DB, HTTP, FS)
│   ├── database.rs   # SQLx/SeaORM implementations
│   ├── http_client.rs
│   └── file_storage.rs
├── application/      # Use cases, команды, запросы
│   ├── commands.rs
│   ├── queries.rs
│   └── dto.rs
├── presentation/     # Входные точки (CLI, HTTP, gRPC)
│   ├── cli.rs
│   ├── http/
│   │   ├── routes.rs
│   │   ├── handlers.rs
│   │   └── middleware.rs
│   └── grpc/
└── config.rs         # Конфигурация
```

### Dependency Inversion (Инверсия зависимостей)
```rust
// Domain layer — определяет контракты (трейты)
pub trait UserRepository: Send + Sync {
    fn find_by_id(&self, id: UserId) -> Result<Option<User>, AppError>;
    fn save(&self, user: &User) -> Result<(), AppError>;
}

// Application layer — использует трейты
pub struct UserService<R: UserRepository> {
    repo: R,
}

impl<R: UserRepository> UserService<R> {
    pub fn new(repo: R) -> Self { Self { repo } }
    pub fn get_user(&self, id: UserId) -> Result<User, AppError> {
        self.repo.find_by_id(id)?.ok_or(AppError::NotFound)
    }
}

// Infrastructure layer — реализует трейты
pub struct PostgresUserRepository { pool: PgPool }

impl UserRepository for PostgresUserRepository { ... }
```

## 2. Repository Pattern

### Трейт репозитория
```rust
use async_trait::async_trait;

#[async_trait]
pub trait UserRepository: Send + Sync {
    async fn find_by_id(&self, id: UserId) -> Result<Option<User>, RepoError>;
    async fn find_by_email(&self, email: &str) -> Result<Option<User>, RepoError>;
    async fn save(&self, user: &User) -> Result<(), RepoError>;
    async fn delete(&self, id: UserId) -> Result<(), RepoError>;
    async fn list(&self, filter: UserFilter) -> Result<Vec<User>, RepoError>;
}
```

### Реализация для SQLx
```rust
pub struct SqlxUserRepository {
    pool: PgPool,
}

#[async_trait]
impl UserRepository for SqlxUserRepository {
    async fn find_by_id(&self, id: UserId) -> Result<Option<User>, RepoError> {
        let row = sqlx::query_as!(
            UserRow,
            "SELECT id, email, name, created_at FROM users WHERE id = $1",
            id.0
        )
        .fetch_optional(&self.pool)
        .await?;
        Ok(row.map(User::from))
    }
    // ... остальные методы
}
```

### In-memory для тестов
```rust
pub struct InMemoryUserRepository {
    users: RwLock<HashMap<UserId, User>>,
}

#[async_trait]
impl UserRepository for InMemoryUserRepository {
    async fn find_by_id(&self, id: UserId) -> Result<Option<User>, RepoError> {
        Ok(self.users.read().await.get(&id).cloned())
    }
    // ... просто работа с HashMap
}
```

## 3. Service Layer (Бизнес-логика)

### Use Case / Command
```rust
#[derive(Debug, Validate)]
pub struct CreateUserCommand {
    #[validate(email)]
    pub email: String,
    #[validate(length(min = 2))]
    pub name: String,
}

pub struct CreateUserUseCase<R: UserRepository> {
    repo: R,
    email_sender: EmailSender,
}

impl<R: UserRepository> CreateUserUseCase<R> {
    pub async fn execute(&self, cmd: CreateUserCommand) -> Result<User, AppError> {
        // Валидация
        cmd.validate()?;

        // Проверка уникальности
        if self.repo.find_by_email(&cmd.email).await?.is_some() {
            return Err(AppError::Conflict("Email already exists"));
        }

        // Создание
        let user = User::new(cmd.email, cmd.name);
        self.repo.save(&user).await?;
        self.email_sender.send_welcome(&user).await?;

        Ok(user)
    }
}
```

### Query (чтение без изменений)
```rust
pub struct GetUserQuery<R: UserRepository> {
    repo: R,
}

impl<R: UserRepository> GetUserQuery<R> {
    pub async fn execute(&self, id: UserId) -> Result<UserDto, AppError> {
        let user = self.repo.find_by_id(id).await?
            .ok_or(AppError::NotFound)?;
        Ok(UserDto::from(user))
    }
}
```

## 4. DTO (Data Transfer Objects)

### Разделение Domain и DTO
```rust
// Domain entity — богатая, с методами
pub struct User {
    id: UserId,
    email: Email,
    name: String,
    password_hash: String,
    created_at: DateTime<Utc>,
    version: u64, // для optimistic locking
}

impl User {
    pub fn new(email: String, name: String) -> Self { ... }
    pub fn change_email(&mut self, new: Email) -> Result<(), DomainError> { ... }
    pub fn verify_password(&self, password: &str) -> bool { ... }
}

// DTO для API — простая, сериализуемая
#[derive(Serialize, Deserialize)]
pub struct UserDto {
    pub id: Uuid,
    pub email: String,
    pub name: String,
    pub created_at: DateTime<Utc>,
}

impl From<User> for UserDto {
    fn from(user: User) -> Self {
        Self { id: user.id.0, email: user.email.0, name: user.name, created_at: user.created_at }
    }
}

// Request/Response DTO
#[derive(Deserialize, Validate)]
pub struct CreateUserRequest {
    #[validate(email)]
    pub email: String,
    #[validate(length(min = 2))]
    pub name: String,
    #[validate(length(min = 8))]
    pub password: String,
}

#[derive(Serialize)]
pub struct CreateUserResponse {
    pub user: UserDto,
}
```

```rust
use config::{Config, File, Environment};

    pub workers: usize,

#[derive(Deserialize, Clone)]
pub struct DatabaseConfig {
    pub url: String,
    pub max_connections: u32,
    pub min_connections: u32,

    pub fn load() -> Result<Self, config::ConfigError> {
        let mut cfg = Config::builder()
            .add_source(File::with_name("config/default"))

            .add_source(Environment::with_prefix("APP").separator("__"))
            .build()?;
        cfg.try_deserialize()
    }
}
```

## 6. Обработка ошибок приложения

### AppError — единая ошибка
```rust
use thiserror::Error;

#[derive(Error, Debug)]
pub enum AppError {
    #[error("Not found: {0}")]
    NotFound(String),

    #[error("Conflict: {0}")]
    Conflict(String),

    #[error("Validation failed: {0}")]
    Validation(String),

    #[error("Unauthorized")]
    Unauthorized,

    #[error("Forbidden")]
    Forbidden,

    #[error("Internal error: {0}")]
    Internal(#[from] anyhow::Error),

    #[error("Database error: {0}")]
    Database(#[from] sqlx::Error),

    #[error("External service error: {0}")]
    External(String),
}

// Преобразование в HTTP ответ
impl IntoResponse for AppError {
    fn into_response(self) -> Response {
        let (status, message) = match self {
            AppError::NotFound(msg) => (StatusCode::NOT_FOUND, msg),
            AppError::Conflict(msg) => (StatusCode::CONFLICT, msg),
            AppError::Validation(msg) => (StatusCode::BAD_REQUEST, msg),
            AppError::Unauthorized => (StatusCode::UNAUTHORIZED, "Unauthorized".into()),
            AppError::Forbidden => (StatusCode::FORBIDDEN, "Forbidden".into()),
            _ => (StatusCode::INTERNAL_SERVER_ERROR, "Internal error".into()),
        };
        (status, Json(ErrorResponse { error: message })).into_response()
    }
}
```

## 6. Dependency Injection (DI) контейнер

### Ручной DI (простой и явный)
```rust
pub struct AppState {
    pub user_repo: Arc<dyn UserRepository>,
    pub order_repo: Arc<dyn OrderRepository>,
    pub email_sender: Arc<dyn EmailSender>,
    pub config: AppConfig,
}

impl AppState {
    pub async fn new(config: AppConfig) -> Result<Self, AppError> {
        let pool = PgPool::connect(&config.database.url).await?;
        let user_repo = Arc::new(SqlxUserRepository::new(pool.clone()));
        let order_repo = Arc::new(SqlxOrderRepository::new(pool));
        let email_sender = Arc::new(SmtpEmailSender::new(&config.email));

        Ok(Self { user_repo, order_repo, email_sender, config })
    }
}
```

### Использование в handlers
```rust
async fn create_user(
    State(state): State<Arc<AppState>>,
    Json(req): Json<CreateUserRequest>,
) -> Result<Json<CreateUserResponse>, AppError> {
    let use_case = CreateUserUseCase::new(
        state.user_repo.clone(),
        state.email_sender.clone(),
    );
    let user = use_case.execute(req.into()).await?;
    Ok(Json(CreateUserResponse { user: user.into() }))
}
```

## 7. Middleware и Cross-cutting concerns

### Logging middleware
```rust
async fn logging_middleware(
    request: Request,
    next: Next,
) -> Response {
    let start = Instant::now();
    let method = request.method().clone();
    let uri = request.uri().clone();

    let response = next.run(request).await;

    let latency = start.elapsed();
    info!("{} {} - {} - {:?}", method, uri, response.status(), latency);
    response
}
```

### Auth middleware
```rust
async fn auth_middleware(
    State(state): State<Arc<AppState>>,
    mut request: Request,
    next: Next,
) -> Result<Response, AppError> {
    let token = request
        .headers()
        .get("Authorization")
        .and_then(|h| h.to_str().ok())
        .and_then(|h| h.strip_prefix("Bearer "));

    let user_id = token
        .ok_or(AppError::Unauthorized)?
        .verify(&state.config.jwt_secret)?;

    request.extensions_mut().insert(user_id);
    Ok(next.run(request).await)
}
```

## 11. Типичные ошибки архитектуры

| Проблема | Признак | Решение |
|----------|---------|---------|
| God Object | Один сервис делает всё | Разбить на мелкие Use Cases |
| Anemic Domain | Сущности — просто данные | Перенести логику в методы Entity |
| Leaky Abstraction | SQL в Service layer | Repository pattern |
| Tight Coupling | Конкретные типы в сигнатурах | Использовать трейты (DI) |
| Untestable | Прямые вызовы DB/HTTP в коде | Моки через трейты |
| Config sprawl | `std::env::var` по всему коду | Централизованный `Config` |
