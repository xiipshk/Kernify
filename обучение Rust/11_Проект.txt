## Требования к приложению

Консольное приложение для управления задачами с командами:
- `add "text"` — добавить задачу
- `list` — показать все задачи
- `done <id>` — отметить выполненной
- `remove <id>` — удалить
- `help` — справка
- `quit` — выход

Данные сохраняются в JSON файл.

## src/models/task.rs

```rust
use serde::{Deserialize, Serialize};
use std::fmt;

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct Task {
    pub id: u64,
    pub text: String,
    pub completed: bool,
    pub created_at: String,
}

impl Task {
    pub fn new(id: u64, text: String) -> Self {
        use chrono::Local;
        Task {
            id,
            text,
            completed: false,
            created_at: Local::now().format("%Y-%m-%d %H:%M").to_string(),
        }
    }
}

impl fmt::Display for Task {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        let status = if self.completed { "✓" } else { "○" };
        write!(f, "[{}] {}: {} ({})", status, self.id, self.text, self.created_at)
    }
}
```

## src/storage/json_storage.rs

```rust
use anyhow::Result;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::fs;
use std::path::Path;

use crate::models::Task;

#[derive(Serialize, Deserialize, Default)]
pub struct TaskStore {
    tasks: HashMap<u64, Task>,
    next_id: u64,
}

pub struct JsonStorage {
    path: String,
}

impl JsonStorage {
    pub fn new(path: impl Into<String>) -> Self {
        Self { path: path.into() }
    }

    fn load(&self) -> Result<TaskStore> {
        if !Path::new(&self.path).exists() {
            return Ok(TaskStore::default());
        }
        let content = fs::read_to_string(&self.path)?;
        if content.trim().is_empty() {
            return Ok(TaskStore::default());
        }
        Ok(serde_json::from_str(&content)?)
    }

    fn save(&self, store: &TaskStore) -> Result<()> {
        let content = serde_json::to_string_pretty(store)?;
        fs::write(&self.path, content)?;
        Ok(())
    }

    pub fn get_all(&self) -> Result<Vec<Task>> {
        let store = self.load()?;
        let mut tasks: Vec<_> = store.tasks.into_values().collect();
        tasks.sort_by_key(|t| t.id);
        Ok(tasks)
    }

    pub fn add(&self, text: String) -> Result<Task> {
        let mut store = self.load()?;
        let task = Task::new(store.next_id, text);
        store.tasks.insert(store.next_id, task.clone());
        store.next_id += 1;
        self.save(&store)?;
        Ok(task)
    }

    pub fn complete(&self, id: u64) -> Result<()> {
        let mut store = self.load()?;
        if let Some(task) = store.tasks.get_mut(&id) {
            task.completed = true;
            self.save(&store)?;
            Ok(())
        } else {
            Err(anyhow::anyhow!("Task {} not found", id))
        }
    }

    pub fn remove(&self, id: u64) -> Result<()> {
        let mut store = self.load()?;
        if store.tasks.remove(&id).is_some() {
            self.save(&store)?;
            Ok(())
        } else {
            Err(anyhow::anyhow!("Task {} not found", id))
        }
    }
}
```

## src/commands/mod.rs

```rust
pub mod add;
pub mod list;
pub mod done;
pub mod remove;

use anyhow::Result;
use crate::storage::JsonStorage;

pub enum Command {
    Add(String),
    List,
    Done(u64),
    Remove(u64),
    Help,
    Quit,
}

pub fn parse(input: &str) -> Option<Command> {
    let parts: Vec<_> = input.split_whitespace().collect();
    match parts.as_slice() {
        ["add", rest @ ..] if !rest.is_empty() => Some(Command::Add(rest.join(" "))),
        ["list"] => Some(Command::List),
        ["done", id] => id.parse().ok().map(Command::Done),
        ["remove" | "rm", id] => id.parse().ok().map(Command::Remove),
        ["help"] => Some(Command::Help),
        ["quit" | "exit"] => Some(Command::Quit),
        _ => None,
    }
}

pub fn execute(cmd: Command, storage: &JsonStorage) -> Result<bool> {
    match cmd {
        Command::Add(text) => add::execute(storage, text).map(|_| true),
        Command::List => list::execute(storage).map(|_| true),
        Command::Done(id) => done::execute(storage, id).map(|_| true),
        Command::Remove(id) => remove::execute(storage, id).map(|_| true),
        Command::Help => { print_help(); Ok(true) }
        Command::Quit => Ok(false),
    }
}

fn print_help() {
    println!(r#"
Todo CLI — управление задачами

Commands:
  add "text"     — добавить задачу
  list           — показать все задачи
  done <id>      — отметить выполненной
  remove <id>    — удалить задачу
  help           — эта справка
  quit           — выход
"#);
}
```

## src/main.rs

```rust
use anyhow::Result;
use todo_cli::{JsonStorage, Command, parse, execute};
use std::io::{self, Write};

const STORAGE_PATH: &str = "tasks.json";

fn main() -> Result<()> {
    let storage = JsonStorage::new(STORAGE_PATH);
    println!("Todo CLI — type 'help' for commands");

    loop {
        print!("> ");
        io::stdout().flush()?;

        let mut input = String::new();
        io::stdin().read_line(&mut input)?;
        let input = input.trim();

        if input.is_empty() { continue; }

        match parse(input) {
            Some(cmd) => {
                let continue_loop = execute(cmd, &storage)?;
                if !continue_loop { break; }
            }
            None => println!("Unknown command. Type 'help' for usage."),
        }
    }

    println!("Goodbye!");
    Ok(())
}
```
