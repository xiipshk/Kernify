## 1. JSON — стандарт обмена данными

### Стандартная библиотека
```python
import json

data = {"name": "Анна", "age": 25, "skills": ["Python", "SQL"]}

# В строку
json_str = json.dumps(data, ensure_ascii=False, indent=2)
# '{\n  "name": "Анна",\n  "age": 25,\n  "skills": ["Python", "SQL"]\n}'

# Из строки
parsed = json.loads(json_str)

# В файл
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)

# Из файла
with open("data.json", encoding="utf-8") as f:
    data = json.load(f)
```

### Кастомный энкодер
```python
from datetime import datetime
from dataclasses import asdict

class CustomEncoder(json.JSONEncoder):
    def default(self, obj):
        if isinstance(obj, datetime):
            return obj.isoformat()
        if hasattr(obj, "__dataclass_fields__"):
            return asdict(obj)
        return super().default(obj)

json.dumps({"time": datetime.now(), "user": User("Анна")}, cls=CustomEncoder)
```

## 2. CSV — табличные данные

### Стандартная библиотека
```python
import csv

# Запись
with open("users.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["name", "age", "city"])
    writer.writerow(["Анна", 25, "Москва"])
    writer.writerow(["Борис", 30, "СПб"])

# DictWriter
with open("users.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.DictWriter(f, fieldnames=["name", "age", "city"])
    writer.writeheader()
    writer.writerow({"name": "Анна", "age": 25, "city": "Москва"})

# Чтение
with open("users.csv", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(row)  # {'name': 'Анна', 'age': '25', 'city': 'Москва'}
```

## 4. SQLite — встроенная БД

### sqlite3 — низкоуровневый доступ
```python
import sqlite3

conn = sqlite3.connect("app.db")
conn.row_factory = sqlite3.Row  # доступ по имени колонки
cursor = conn.cursor()

# Создание таблицы
cursor.execute("""
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        email TEXT UNIQUE,
        age INTEGER
    )
""")

# Вставка (параметризованный запрос!)
cursor.execute("INSERT INTO users (name, email, age) VALUES (?, ?, ?)",
               ("Анна", "anna@example.com", 25))
conn.commit()

# Выборка
cursor.execute("SELECT * FROM users WHERE age > ?", (18,))
for row in cursor.fetchall():
    print(dict(row))  # {'id': 1, 'name': 'Анна', ...}

conn.close()
```

### Контекстный менеджер
```python
with sqlite3.connect("app.db") as conn:
    conn.execute("INSERT INTO users (name) VALUES (?)", ("Борис",))
    # авто-commit при выходе
```

## 5. SQLAlchemy — ORM и Core

### Установка
```bash
pip install sqlalchemy
# Для async
pip install sqlalchemy[asyncio] asyncpg  # PostgreSQL
pip install sqlalchemy[asyncio] aiosqlite  # SQLite async
```

### Declarative ORM (2.0 стиль)
```python
from sqlalchemy import create_engine, Column, Integer, String, select
from sqlalchemy.orm import DeclarativeBase, Session, Mapped, mapped_column

class Base(DeclarativeBase):
    pass

class User(Base):
    __tablename__ = "users"
    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    name: Mapped[str] = mapped_column(String(50))
    email: Mapped[str] = mapped_column(String(100), unique=True)
    age: Mapped[int] = mapped_column(Integer, default=0)

engine = create_engine("sqlite:///app.db", echo=True)
Base.metadata.create_all(engine)

# Работа с сессией
with Session(engine) as session:
    user = User(name="Анна", email="anna@example.com", age=25)
    session.add(user)
    session.commit()

    # Запросы
    users = session.scalars(select(User).where(User.age > 18)).all()
    for u in users:
        print(u.name, u.email)
```

### Асинхронный SQLAlchemy
```python
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
from sqlalchemy.orm import sessionmaker

async_engine = create_async_engine("sqlite+aiosqlite:///app.db", echo=True)
async_session = sessionmaker(async_engine, class_=AsyncSession, expire_on_commit=False)

async def get_users():
    async with async_session() as session:
        result = await session.execute(select(User).where(User.age > 18))
        return result.scalars().all()
```

### Core — SQL Expression Language
```python
from sqlalchemy import Table, MetaData, insert, select, update, delete

metadata = MetaData()
users = Table("users", metadata, autoload_with=engine)

with engine.connect() as conn:
    # Insert
    conn.execute(insert(users).values(name="Виктор", age=35))
    conn.commit()

    # Select
    result = conn.execute(select(users.c.name, users.c.age).where(users.c.age > 20))
    for row in result:
        print(row)
```

## 7. Сериализация объектов

### pickle — встроенный (НЕ для недоверенных данных!)
```python
import pickle

data = {"users": [{"name": "Анна"}, {"name": "Борис"}]}
pickled = pickle.dumps(data)
restored = pickle.loads(pickled)

# Протокол 5 (Python 3.8+) — быстрее для больших объектов
pickle.dumps(data, protocol=pickle.HIGHEST_PROTOCOL)
```

### msgpack — бинарный, быстрый, кроссплатформенный
```bash
pip install msgpack
```

```python
import msgpack

data = {"key": "value", "numbers": [1, 2, 3]}
packed = msgpack.packb(data, use_bin_type=True)
unpacked = msgpack.unpackb(packed, raw=False)
```

### Pydantic — валидация + сериализация
```bash
pip install pydantic
```

```python
from pydantic import BaseModel, EmailStr
from typing import Optional
from datetime import datetime

class User(BaseModel):
    id: int
    name: str
    email: EmailStr
    age: Optional[int] = None
    created_at: datetime = datetime.now()

user = User(id=1, name="Анна", email="anna@example.com")
print(user.model_dump())      # dict
print(user.model_dump_json()) # JSON string

# Парсинг с валидацией
user2 = User.model_validate({"id": "2", "name": "Борис", "email": "boris@test.com"})
```

## Типичные ошибки

| Ошибка | Правильно |
|--------|-----------|
| SQL-инъекция: `f"SELECT * FROM t WHERE x={val}"` | Параметризованные запросы: `execute("... WHERE x=?", (val,))` |
| `yaml.load()` без `safe_load` | Всегда `yaml.safe_load()` |
| pickle недоверенных данных | Никогда! Используйте JSON/msgpack |
| Открытые соединения не закрыты | `with` контекстные менеджеры / `session.close()` |
| N+1 проблема в ORM | `selectinload`, `joinedload` для связанных данных |
