# Урок 10: Проект для начинающих — «Список задач (Todo App)»

## Цели проекта
- Закрепить все темы уроков 1-9 на практике
- Научиться структурировать небольшое приложение
- Понять разделение: данные (State), представление (View), логика (Controller)
- Получить работающее приложение в портфолио

---

## Требования к приложению

**Функционал (MVP):**
1. Добавление задачи (Enter или кнопка)
2. Отметка выполненной (чекбокс)
3. Удаление задачи (кнопка ×)
4. Фильтры: Все / Активные / Выполненные
5. Счётчик оставшихся задач
6. Кнопка «Очистить выполненные»
7. Сохранение в localStorage (данные не пропадают после перезагрузки)

**Дополнительно (Bonus):**
- Редактирование задачи по двойному клику
- Drag & Drop для переупорядочивания
- Анимации появления/удаления
- Горячие клавиши (Esc — отмена редактирования)

## 3. JavaScript (app.js) — Модель

```javascript
// ==================== MODEL ====================

const STORAGE_KEY = 'todos-vanilla-js';

/** Хранилище задач */
class TodoModel {
  constructor() {
    this.todos = this.load();
    this.listeners = new Set();
  }

  /** Подписка на изменения */
  subscribe(listener) {
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }

  /** Уведомить слушателей */
  notify() {
    this.listeners.forEach(fn => fn(this.todos));
  }

  /** Загрузка из localStorage */
  load() {
    try {
      const data = localStorage.getItem(STORAGE_KEY);
      return data ? JSON.parse(data) : [];
    } catch {
      return [];
    }
  }

  /** Сохранение в localStorage */
  save() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(this.todos));
  }

  /** Генерация ID */
  generateId() {
    return Date.now() + Math.random().toString(36).slice(2, 9);
  }

  /** Получить все */
  getAll() {
    return [...this.todos];
  }

  /** Добавить задачу */
  add(text) {
    const trimmed = text.trim();
    if (!trimmed) return null;

    const todo = {
      id: this.generateId(),
      text: trimmed,
      completed: false,
      createdAt: Date.now()
    };

    this.todos.push(todo);
    this.save();
    this.notify();
    return todo;
  }

  /** Переключить статус */
  toggle(id) {
    const todo = this.todos.find(t => t.id === id);
    if (todo) {
      todo.completed = !todo.completed;
      this.save();
      this.notify();
    }
  }

  /** Удалить задачу */
  remove(id) {
    const index = this.todos.findIndex(t => t.id === id);
    if (index !== -1) {
      this.todos.splice(index, 1);
      this.save();
      this.notify();
    }
  }

  /** Очистить выполненные */
  clearCompleted() {
    this.todos = this.todos.filter(t => !t.completed);
    this.save();
    this.notify();
  }

  /** Переключить все */
  toggleAll(completed) {
    let changed = false;
    this.todos.forEach(t => {
      if (t.completed !== completed) {
        t.completed = completed;
        changed = true;
      }
    });
    if (changed) {
      this.save();
      this.notify();
    }
  }

  /** Обновить текст */
  updateText(id, text) {
    const trimmed = text.trim();
    const todo = this.todos.find(t => t.id === id);
    if (todo && trimmed) {
      todo.text = trimmed;
      this.save();
      this.notify();
      return true;
    }
    return false;
  }

  /** Получить отфильтрованные */
  getFiltered(filter) {
    switch (filter) {
      case 'active':
        return this.todos.filter(t => !t.completed);
      case 'completed':
        return this.todos.filter(t => t.completed);
      default:
        return [...this.todos];
    }
  }

  /** Счётчик активных */
  getActiveCount() {
    return this.todos.filter(t => !t.completed).length;
  }

  /** Есть ли выполненные */
  hasCompleted() {
    return this.todos.some(t => t.completed);
  }

  /** Все ли выполнены */
  allCompleted() {
    return this.todos.length > 0 && this.todos.every(t => t.completed);
  }
}
```

## 4. JavaScript (app.js) — View

```javascript
// ==================== VIEW ====================

class TodoView {
  constructor() {
    // Кэш DOM элементов
    this.elements = {
      newTodo: document.getElementById('new-todo'),
      todoList: document.getElementById('todo-list'),
      main: document.querySelector('.main'),
      footer: document.querySelector('.footer'),
      toggleAll: document.getElementById('toggle-all'),
      todoCount: document.getElementById('todo-count'),
      clearCompleted: document.getElementById('clear-completed'),
      filters: document.querySelectorAll('.filter')
    };

    // Текущий фильтр
    this.currentFilter = 'all';

    // Callback'и (устанавливаются контроллером)
    this.onAddTodo = null;
    this.onToggleTodo = null;
    this.onRemoveTodo = null;
    this.onToggleAll = null;
    this.onClearCompleted = null;
    this.onFilterChange = null;
    this.onEditTodo = null;
    this.onCancelEdit = null;
  }

  /** Привязка событий */
  bindEvents() {
    // Добавление
    this.elements.newTodo.addEventListener('keypress', e => {
      if (e.key === 'Enter' && this.onAddTodo) {
        this.onAddTodo(e.target.value);
        e.target.value = '';
      }
    });

    // Делегирование для списка
    this.elements.todoList.addEventListener('click', e => {
      const item = e.target.closest('.todo-item');
      if (!item) return;

      const id = item.dataset.id;

      // Чекбокс
      if (e.target.matches('.toggle')) {
        this.onToggleTodo?.(id);
      }
      // Удаление
      else if (e.target.matches('.destroy')) {
        this.onRemoveTodo?.(id);
      }
      // Двойной клик — редактирование
      else if (e.target.matches('.todo-text')) {
        this.startEditing(item, id);
      }
    });

    // Редактирование — клавиши
    this.elements.todoList.addEventListener('keydown', e => {
      if (e.target.matches('.edit-input')) {
        if (e.key === 'Enter') {
          this.finishEditing(e.target);
        } else if (e.key === 'Escape') {
          this.cancelEditing(e.target);
        }
      }
    });

    // Потеря фокуса — сохранение
    this.elements.todoList.addEventListener('focusout', e => {
      if (e.target.matches('.edit-input')) {
        this.finishEditing(e.target);
      }
    });

    // Toggle all
    this.elements.toggleAll.addEventListener('change', e => {
      this.onToggleAll?.(e.target.checked);
    });

    // Clear completed
    this.elements.clearCompleted.addEventListener('click', () => {
      this.onClearCompleted?.();
    });

    // Фильтры
    this.elements.filters.forEach(link => {
      link.addEventListener('click', e => {
        e.preventDefault();
        const filter = e.target.dataset.filter;
        this.setFilter(filter);
        this.onFilterChange?.(filter);
      });
    });
  }

  /** Рендер списка */
  render(todos) {
    this.elements.todoList.innerHTML = '';

    todos.forEach(todo => {
      const li = this.createTodoElement(todo);
      this.elements.todoList.append(li);
    });

    this.updateUI(todos);
  }

  /** Создать элемент задачи */
  createTodoElement(todo) {
    const li = document.createElement('li');
    li.className = `todo-item${todo.completed ? ' completed' : ''}`;
    li.dataset.id = todo.id;

    li.innerHTML = `
      <input class="toggle" type="checkbox" ${todo.completed ? 'checked' : ''}>
      <label class="todo-text">${this.escapeHtml(todo.text)}</label>
      <input class="edit-input" type="text" value="${this.escapeHtml(todo.text)}" style="display:none">
      <button class="destroy" aria-label="Удалить">×</button>
    `;

    return li;
  }

  /** Обновление счётчиков и видимости */
  updateUI(todos) {
    const activeCount = todos.filter(t => !t.completed).length;
    const hasCompleted = todos.some(t => t.completed);
    const allCompleted = todos.length > 0 && todos.every(t => t.completed);

    // Счётчик
    this.elements.todoCount.innerHTML = `
      <strong>${activeCount}</strong> ${this.pluralize(activeCount, 'задача', 'задачи', 'задач')} осталось
    `;

    // Toggle all
    this.elements.toggleAll.checked = allCompleted;

    // Показать/скрыть main и footer
    this.elements.main.hidden = todos.length === 0;
    this.elements.footer.hidden = todos.length === 0;

    // Clear completed
    this.elements.clearCompleted.hidden = !hasCompleted;

    // Активный фильтр
    this.elements.filters.forEach(link => {
      link.classList.toggle('selected', link.dataset.filter === this.currentFilter);
    });
  }

  /** Установить активный фильтр */
  setFilter(filter) {
    this.currentFilter = filter;
    this.elements.filters.forEach(link => {
      link.classList.toggle('selected', link.dataset.filter === filter);
    });
  }

  /** Начало редактирования */
  startEditing(item, id) {
    item.classList.add('editing');
    const input = item.querySelector('.edit-input');
    input.style.display = 'block';
    input.focus();
    input.setSelectionRange(input.value.length, input.value.length);
  }

  /** Завершение редактирования */
  finishEditing(input) {
    const item = input.closest('.todo-item');
    const id = item.dataset.id;
    const text = input.value;

    item.classList.remove('editing');
    input.style.display = 'none';

    if (this.onEditTodo) {
      this.onEditTodo(id, text);
    }
  }

  /** Отмена редактирования */
  cancelEditing(input) {
    const item = input.closest('.todo-item');
    item.classList.remove('editing');
    input.style.display = 'none';
    this.onCancelEdit?.();
  }

  /** Экранирование HTML */
  escapeHtml(text) {
    const div = document.createElement('div');
    div.textContent = text;
    return div.innerHTML;
  }

  /** Склонение слов */
  pluralize(n, one, two, five) {
    const mod10 = n % 10;
    const mod100 = n % 100;
    if (mod10 === 1 && mod100 !== 11) return one;
    if (mod10 >= 2 && mod10 <= 4 && (mod100 < 10 || mod100 >= 20)) return two;
    return five;
  }

  /** Очистка поля ввода */
  clearInput() {
    this.elements.newTodo.value = '';
    this.elements.newTodo.focus();
  }
}
```

```javascript
// ==================== CONTROLLER ====================

class TodoController {
  constructor(model, view) {
    this.model = model;
    this.view = view;

    // Привязка callback'ов
    view.onAddTodo = this.handleAddTodo.bind(this);
    view.onToggleTodo = this.handleToggleTodo.bind(this);
    view.onRemoveTodo = this.handleRemoveTodo.bind(this);
    view.onToggleAll = this.handleToggleAll.bind(this);
    view.onClearCompleted = this.handleClearCompleted.bind(this);
    view.onFilterChange = this.handleFilterChange.bind(this);
    view.onEditTodo = this.handleEditTodo.bind(this);
    view.onCancelEdit = this.handleCancelEdit.bind(this);

    // Подписка на изменения модели
    this.unsubscribe = model.subscribe(this.onModelChange.bind(this));

    // Инициализация
    view.bindEvents();
    this.render();
  }

  /** Модель изменилась — перерисовать */
  onModelChange(todos) {
    this.render();
  }

  /** Рендер с текущим фильтром */
  render() {
    const filtered = this.model.getFiltered(this.view.currentFilter);
    this.view.render(filtered);
  }

  // --- Handlers ---

  handleAddTodo(text) {
    this.model.add(text);
  }

  handleToggleTodo(id) {
    this.model.toggle(id);
  }

  handleRemoveTodo(id) {
    this.model.remove(id);
  }

  handleToggleAll(completed) {
    this.model.toggleAll(completed);
  }

  handleClearCompleted() {
    this.model.clearCompleted();
  }

  handleFilterChange(filter) {
    this.render();
  }

  handleEditTodo(id, text) {
    this.model.updateText(id, text);
  }

  handleCancelEdit() {
    // просто перерисовать (вернёт старый текст)
    this.render();
  }

  /** Очистка при удалении приложения */
  destroy() {
    this.unsubscribe();
  }
}

// ==================== INIT ====================

// Ждём готовности DOM
document.addEventListener('DOMContentLoaded', () => {
  const model = new TodoModel();
  const view = new TodoView();
  const controller = new TodoController(model, view);

  // Глобальный доступ для отладки в консоли
  window.__todoApp = { model, view, controller };
});
```

## Разбор ключевых моментов

### Разделение ответственности
- **Model** — только данные и localStorage, ничего не знает про DOM
- **View** — только DOM манипуляции, не знает про localStorage
- **Controller** — связывает их, содержит бизнес-логику

### Делегирование событий
Один слушатель на `ul.todo-list` вместо по слушателю на каждый `li`.

### localStorage
- `JSON.stringify/parse` для сериализации
- `try/catch` на случай поврежденных данных

### Безопасность
- `escapeHtml` при выводе пользовательского текста
- `textContent` вместо `innerHTML` где возможно
