## Требования

**Функционал:**
1. **Авторизация**: Регистрация, вход, JWT в HttpOnly cookie, защищённые роуты
2. **Проекты**: CRUD, переключение между проектами
3. **Задачи**: CRUD, статусы (Backlog/Todo/In Progress/Done), приоритеты, дедлайны
4. **Drag & Drop**: Перемещение задач между колонками (Kanban)
5. **Фильтры/Поиск**: По статусу, приоритету, тексту, тегам
6. **Real-time** (опционально): WebSocket уведомления об изменениях
7. **Офлайн**: Service Worker, IndexedDB кэш, синхронизация при онлайне

**Технологический стек:**
- **Билд**: Vite + TypeScript
- **Роутинг**: Собственный роутер (History API) или wouter
- **Состояние**: Собственный стор (EventEmitter + Proxy) или Zustand
- **API**: Fetch wrapper с interceptors, retry, токены
- **Стили**: CSS Modules / PostCSS / Tailwind (на выбор)
- **Тесты**: Vitest + Testing Library + MSW (Mock Service Worker)
- **Линт**: ESLint + Prettier + Husky
- **CI**: GitHub Actions (lint, test, build, e2e)

### 1. API Client (shared/api/client.ts)

```typescript
type RequestOptions = RequestInit & {
  params?: Record<string, string>;
  skipAuth?: boolean;
  retry?: number;
};

class ApiClient {
  private baseUrl: string;
  private refreshPromise: Promise<string> | null = null;

  constructor(baseUrl: string) {
    this.baseUrl = baseUrl;
  }

  private async getAccessToken(): Promise<string | null> {
    // Токен в памяти или из cookie (HttpOnly — читать нельзя, но отправляется автоматически)
    return getCookie('access_token');
  }

  private async refreshToken(): Promise<string> {
    if (!this.refreshPromise) {
      this.refreshPromise = (async () => {
        const res = await fetch(`${this.baseUrl}/auth/refresh`, {
          method: 'POST',
          credentials: 'include'
        });
        if (!res.ok) throw new Error('Refresh failed');
        const { accessToken } = await res.json();
        return accessToken;
      })().finally(() => { this.refreshPromise = null; });
    }
    return this.refreshPromise;
  }

  async request<T>(endpoint: string, options: RequestOptions = {}): Promise<T> {
    const { params, skipAuth, retry = 1, headers, ...init } = options;

    const url = new URL(endpoint, this.baseUrl);
    if (params) Object.entries(params).forEach(([k, v]) => url.searchParams.set(k, v));

    const requestHeaders: HeadersInit = {
      'Content-Type': 'application/json',
      ...headers
    };

    if (!skipAuth) {
      const token = await this.getAccessToken();
      if (token) (requestHeaders as Record<string, string>)['Authorization'] = `Bearer ${token}`;
    }

    const makeRequest = async (attempt: number): Promise<Response> => {
      const res = await fetch(url, { ...init, headers: requestHeaders, credentials: 'include' });

      if (res.status === 401 && !skipAuth && attempt < retry) {
        await this.refreshToken();
        return makeRequest(attempt + 1);
      }
      return res;
    };

    const response = await makeRequest(0);

    if (!response.ok) {
      const error = await response.json().catch(() => ({ message: 'Unknown error' }));
      throw new ApiError(response.status, error.message, error);
    }

    if (response.status === 204) return null as T;
    return response.json();
  }

  get<T>(endpoint: string, opts?: RequestOptions) { return this.request<T>(endpoint, { ...opts, method: 'GET' }); }
  post<T>(endpoint: string, body: unknown, opts?: RequestOptions) { return this.request<T>(endpoint, { ...opts, method: 'POST', body: JSON.stringify(body) }); }
  patch<T>(endpoint: string, body: unknown, opts?: RequestOptions) { return this.request<T>(endpoint, { ...opts, method: 'PATCH', body: JSON.stringify(body) }); }
  delete<T>(endpoint: string, opts?: RequestOptions) { return this.request<T>(endpoint, { ...opts, method: 'DELETE' }); }
}

export const api = new ApiClient(import.meta.env.VITE_API_URL || '/api');

export class ApiError extends Error {
  constructor(public status: number, message: string, public data?: unknown) {
    super(message);
    this.name = 'ApiError';
  }
}
```

### 2. Store (shared/store/index.ts)

```typescript
type Listener<T> = (state: T) => void;

export function createStore<T extends object>(initialState: T) {
  let state = initialState;
  const listeners = new Set<Listener<T>>();

  // Proxy для отслеживания мутаций
  const proxy = new Proxy(state, {
    set(target, prop, value) {
      target[prop as keyof T] = value;
      listeners.forEach(fn => fn(state));
      return true;
    }
  });

  return {
    getState: () => state,
    setState: (partial: Partial<T> | ((s: T) => Partial<T>)) => {
      const next = typeof partial === 'function' ? partial(state) : partial;
      state = { ...state, ...next };
      listeners.forEach(fn => fn(state));
    },
    subscribe: (fn: Listener<T>) => {
      listeners.add(fn);
      fn(state);
      return () => listeners.delete(fn);
    },
    // Для React/Vue интеграции
    useStore: (selector?: (s: T) => any) => {
      // Реализация через useSyncExternalStore (React 18+)
    }
  };
}

// Использование
interface AuthState {
  user: User | null;
  isAuthenticated: boolean;
  isLoading: boolean;
}

export const authStore = createStore<AuthState>({
  user: null,
  isAuthenticated: false,
  isLoading: true
});
```

### 3. Роутер (shared/router/index.ts)

```typescript
type RouteHandler = (params: Record<string, string>) => Promise<void> | void;

interface Route {
  path: string;
  handler: RouteHandler;
  guards?: Array<() => boolean | Promise<boolean>>;
}

export class Router {
  private routes: Route[] = [];
  private currentParams: Record<string, string> = {};

  addRoute(path: string, handler: RouteHandler, guards?: Route['guards']) {
    this.routes.push({ path, handler, guards });
  }

  private match(pathname: string): { route: Route; params: Record<string, string> } | null {
    for (const route of this.routes) {
      const regex = this.pathToRegex(route.path);
      const match = pathname.match(regex);
      if (match) {
        const params: Record<string, string> = {};
        match.groups && Object.entries(match.groups).forEach(([k, v]) => params[k] = v!);
        return { route, params };
      }
    }
    return null;
  }

  private pathToRegex(path: string): RegExp {
    const escaped = path
      .replace(/\//g, '\\/')
      .replace(/:(\w+)/g, '(?<$1>[^/]+)');
    return new RegExp(`^${escaped}$`);
  }

  async navigate(path: string, replace = false) {
    const match = this.match(path);
    if (!match) return this.navigate('/404');

    const { route, params } = match;

    // Guards
    for (const guard of route.guards || []) {
      if (!(await guard())) return this.navigate('/login');
    }

    this.currentParams = params;
    replace ? history.replaceState(null, '', path) : history.pushState(null, '', path);
    await route.handler(params);
  }

  start() {
    window.addEventListener('popstate', () => this.navigate(location.pathname, true));
    this.navigate(location.pathname, true);
  }
}

// Глобальный экземпляр
export const router = new Router();

// Хелперы для навигации
export const navigate = (path: string) => router.navigate(path);
export const useParams = () => router.currentParams;
```

### 4. Drag & Drop (features/tasks/dnd.ts)

```typescript
// Нативный HTML5 DnD для Kanban
export function setupKanbanDnD(
  columns: HTMLElement[],
  onMove: (taskId: string, fromColumn: string, toColumn: string, index: number) => void
) {
  let draggedTask: HTMLElement | null = null;
  let sourceColumn: string | null = null;

  columns.forEach(column => {
    const columnId = column.dataset.columnId!;

    column.addEventListener('dragstart', (e) => {
      const task = (e.target as HTMLElement).closest('[data-task-id]') as HTMLElement;
      if (!task) return;

      draggedTask = task;
      sourceColumn = columnId;
      task.classList.add('dragging');
      e.dataTransfer!.effectAllowed = 'move';
    });

    column.addEventListener('dragend', () => {
      if (draggedTask) draggedTask.classList.remove('dragging');
      draggedTask = null;
      sourceColumn = null;
    });

    column.addEventListener('dragover', (e) => {
      e.preventDefault();
      e.dataTransfer!.dropEffect = 'move';

      const afterElement = getDragAfterElement(column, e.clientY);
      const dragging = document.querySelector('.dragging');
      if (afterElement) {
        column.insertBefore(dragging!, afterElement);
      } else {
        column.appendChild(dragging!);
      }
    });

    column.addEventListener('drop', (e) => {
      e.preventDefault();
      if (!draggedTask || !sourceColumn) return;

      const taskId = draggedTask.dataset.taskId!;
      const targetColumn = columnId;
      const tasks = [...column.querySelectorAll('[data-task-id]')];
      const index = tasks.indexOf(draggedTask);

      if (sourceColumn !== targetColumn || index !== -1) {
        onMove(taskId, sourceColumn, targetColumn, index);
      }
    });
  });
}

function getDragAfterElement(container: HTMLElement, y: number): Element | null {
  const draggableElements = [...container.querySelectorAll('[data-task-id]:not(.dragging)')];
  return draggableElements.reduce((closest, child) => {
    const box = child.getBoundingClientRect();
    const offset = y - box.top - box.height / 2;
    if (offset < 0 && offset > closest.offset) {
      return { offset, element: child };
    }
    return closest;
  }, { offset: Number.NEGATIVE_INFINITY, element: null }).element;
}
```

```typescript
// IndexedDB кэш для офлайн-режима
import { openDB } from 'idb';

const DB_NAME = 'task-manager-offline';
const STORE_TASKS = 'tasks';
const STORE_MUTATIONS = 'mutations';

export const offlineDB = await openDB(DB_NAME, 1, {
  upgrade(db) {
    db.createObjectStore(STORE_TASKS, { keyPath: 'id' });
    db.createObjectStore(STORE_MUTATIONS, { keyPath: 'id', autoIncrement: true });
  }
});

// Сохранение задачи в кэш
export async function cacheTask(task: Task) {
  await offlineDB.put(STORE_TASKS, task);
}

// Очередь мутаций для синхронизации
export async function queueMutation(mutation: { type: 'create'|'update'|'delete'; payload: any }) {
  await offlineDB.add(STORE_MUTATIONS, { ...mutation, timestamp: Date.now() });
}

// Синхронизация при возврате онлайн
export async function syncWhenOnline(api: ApiClient) {
  const mutations = await offlineDB.getAll(STORE_MUTATIONS);
  for (const m of mutations) {
    try {
      switch (m.type) {
        case 'create': await api.post('/tasks', m.payload); break;
        case 'update': await api.patch(`/tasks/${m.payload.id}`, m.payload); break;
        case 'delete': await api.delete(`/tasks/${m.payload.id}`); break;
      }
      await offlineDB.delete(STORE_MUTATIONS, m.id);
    } catch (e) {
      console.error('Sync failed', e);
      break; // Остановимся, попробуем позже
    }
  }
}

// Слушатель онлайн/офлайн
window.addEventListener('online', () => syncWhenOnline(api));
```
