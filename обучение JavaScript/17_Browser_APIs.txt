## 2. Events — Система событий

### Фазы события
```
CAPTURING PHASE (вниз) → TARGET PHASE → BUBBLING PHASE (вверх)
window → document → html → body → ... → target → ... → body → html → document → window
```

```javascript
// addEventListener(type, listener, options)
element.addEventListener("click", handler, {
  capture: false,      // true = фаза перехвата (вверх)
  once: false,         // true = удалить после первого вызова
  passive: false,      // true = никогда не вызовет preventDefault()
  signal: abortSignal  // AbortSignal для удаления
});

// Удаление
element.removeEventListener("click", handler, { capture: false });
```

### Объект события
```javascript
element.addEventListener("click", (event) => {
  event.type;           // "click"
  event.target;         // элемент, на котором сработало событие
  event.currentTarget;  // элемент, на котором висит слушатель (=== this)
  event.eventPhase;     // 1=capturing, 2=target, 3=bubbling
  event.bubbles;        // true/false
  event.cancelable;     // true/false
  event.preventDefault(); // отмена действия по умолчанию
  event.stopPropagation(); // остановка всплытия
  event.stopImmediatePropagation(); // остановка + другие слушатели на этом элементе
  event.isTrusted;      // true = пользовательское действие, false = dispatchEvent
});
```

### Делегирование событий
```javascript
// Вместо 100 слушателей на <li> — один на <ul>
ul.addEventListener("click", (e) => {
  const li = e.target.closest("li");
  if (!li || !ul.contains(li)) return; // клик не в li или не наш ul

  const id = li.dataset.id;
  console.log("Clicked item:", id);
});
```

### Кастомные события
```javascript

```

### Полезные события

## 3. Fetch API — HTTP запросы

### Базовое использование
```javascript
// GET
const response = await fetch("/api/users");
if (!response.ok) throw new Error(`HTTP ${response.status}`);
const users = await response.json();

// POST с JSON
const response = await fetch("/api/users", {
  method: "POST",
  headers: {
    "Content-Type": "application/json",
    "Accept": "application/json"
  },
  body: JSON.stringify({ name: "Анна", email: "anna@example.com" })
});

// FormData (multipart/form-data)
const formData = new FormData();
formData.append("name", "Анна");
formData.append("avatar", fileInput.files[0]);
const response = await fetch("/api/upload", {
  method: "POST",
  body: formData // Content-Type ставится автоматически с boundary
});
```

### Работа с ответом
```javascript
const response = await fetch("/api/data");

// Статус
response.ok;        // true если 200-299
response.status;    // 200, 404, 500...
response.statusText; // "OK", "Not Found"

// Заголовки
response.headers.get("Content-Type");
response.headers.has("ETag");
for (const [key, value] of response.headers) {}

// Тело (только ОДИН раз!)
const text = await response.text();
const json = await response.json();
const blob = await response.blob();
const arrayBuffer = await response.arrayBuffer();
const formData = await response.formData();

// Клонирование для повторного чтения
const clone = response.clone();
const json = await clone.json();
```

### Обработка ошибок
```javascript
async function fetchWithErrorHandling(url, options = {}) {
  try {
    const response = await fetch(url, options);

    if (!response.ok) {
      // HTTP ошибки (4xx, 5xx) не выбрасывают исключение!
      const error = new Error(`HTTP ${response.status}: ${response.statusText}`);
      error.status = response.status;
      error.response = response;
      throw error;
    }

    return await response.json();
  } catch (err) {
    if (err.name === "TypeError" && err.message.includes("fetch")) {
      // Сетевая ошибка (нет интернета, CORS, DNS)
      throw new NetworkError("Нет соединения с сервером");
    }
    throw err; // пробрасываем дальше
  }
}
```

### Таймаут и отмена
```javascript
// AbortController (встроенный таймаут)
const controller = new AbortController();
const timeoutId = setTimeout(() => controller.abort(), 5000);

try {
  const response = await fetch("/api/slow", { signal: controller.signal });
  clearTimeout(timeoutId);
} catch (err) {
  if (err.name === "AbortError") console.log("Таймаут!");
}

// AbortSignal.timeout (ES2023)
const response = await fetch("/api/data", {
  signal: AbortSignal.timeout(5000)
});
```

### Параллельные и последовательные запросы
```javascript
// Параллельно — быстро
const [users, posts, comments] = await Promise.all([
  fetch("/api/users").then(r => r.json()),
  fetch("/api/posts").then(r => r.json()),
  fetch("/api/comments").then(r => r.json())
]);

// Последовательно — когда нужен результат предыдущего
const user = await fetch(`/api/users/${id}`).then(r => r.json());
const posts = await fetch(`/api/users/${id}/posts`).then(r => r.json());

// Promise.allSettled — не падает если один упал
const results = await Promise.allSettled(urls.map(u => fetch(u)));
```

### localStorage / sessionStorage
```javascript

```

### IndexedDB — полноценная БД
```javascript
// Открытие БД
const request = indexedDB.open("MyDB", 1);

request.onupgradeneeded = (e) => {
  const db = e.target.result;
  if (!db.objectStoreNames.contains("users")) {
    const store = db.createObjectStore("users", { keyPath: "id", autoIncrement: true });
    store.createIndex("email", "email", { unique: true });
  }
};

request.onsuccess = (e) => {
  const db = e.target.result;
  // Работа с БД...
};

// Запись
const tx = db.transaction("users", "readwrite");
const store = tx.objectStore("users");
store.add({ name: "Анна", email: "anna@example.com" });
tx.oncomplete = () => console.log("Готово");

// Чтение
const tx = db.transaction("users", "readonly");
const store = tx.objectStore("users");
const request = store.get(1);
request.onsuccess = () => console.log(request.result);

// Индексы
const index = store.index("email");
index.get("anna@example.com").onsuccess = (e) => console.log(e.target.result);

// Курсоры для перебора
store.openCursor().onsuccess = (e) => {
  const cursor = e.target.result;
  if (cursor) {
    console.log(cursor.value);
    cursor.continue();
  }
};
```

## 5. URL и History API

### URL и URLSearchParams
```javascript
// Парсинг URL
const url = new URL("https://example.com/path?name=Анна&age=25#section");
url.href;           // полный URL
url.origin;         // "https://example.com"
url.protocol;       // "https:"
url.host;           // "example.com"
url.hostname;       // "example.com"
url.port;           // ""
url.pathname;       // "/path"
url.search;         // "?name=Анна&age=25"
url.hash;           // "#section"
url.username;       // ""
url.password;       // ""

// SearchParams
url.searchParams.get("name");        // "Анна"
url.searchParams.getAll("tag");      // ["js", "ts"]
url.searchParams.has("age");         // true
url.searchParams.set("age", "26");
url.searchParams.append("tag", "react");
url.searchParams.delete("name");
url.searchParams.toString();         // "age=26&tag=js&tag=react&tag=react"
[...url.searchParams];               // [["age", "26"], ["tag", "js"], ...]

// Относительные URL
const relative = new URL("/api/users", "https://example.com");
// https://example.com/api/users
```

### History API — SPA навигация
```javascript
// Добавление записи (меняет URL без перезагрузки)
history.pushState({ page: "profile", userId: 42 }, "", "/profile/42");

// Замена текущей записи
history.replaceState({ page: "settings" }, "", "/settings");

// Навигация
history.back();      // назад
history.forward();   // вперёд
history.go(-2);      // на 2 назад
history.length;      // количество записей
history.state;       // состояние текущей записи

// Событие popstate — назад/вперёд браузера
window.addEventListener("popstate", (e) => {
  console.log(e.state); // { page: "profile", userId: 42 }
  renderPage(e.state);
});

// scrollRestoration
history.scrollRestoration = "manual"; // "auto" по умолчанию
```

### Паттерн SPA роутера
```javascript
function router() {
  const routes = {
    "/": () => renderHome(),
    "/profile/:id": (params) => renderProfile(params.id),
    "/settings": () => renderSettings()
  };

  function match(path) {
    for (const [pattern, handler] of Object.entries(routes)) {
      const regex = patternToRegex(pattern);
      const match = path.match(regex);
      if (match) return { handler, params: match.groups };
    }
    return null;
  }

  function navigate(path) {
    history.pushState({}, "", path);
    const match = match(path);
    if (match) match.handler(match.params);
  }

  window.addEventListener("popstate", () => {
    const match = match(location.pathname);
    if (match) match.handler(match.params);
  });

  return { navigate };
}

const { navigate } = router();
navigate("/profile/42");
```

## 6. Forms — Работа с формами

### FormData
```html
<form id="login-form">
  <input name="email" type="email" required>
  <input name="password" type="password" required>
  <input name="avatar" type="file">
  <button type="submit">Войти</button>
</form>
```

```javascript
const form = document.querySelector("#login-form");

// FormData из формы
const formData = new FormData(form);

// Итерация
for (const [name, value] of formData) {
  console.log(name, value); // value может быть File
}

// Получение значений
formData.get("email");           // первое значение
formData.getAll("tags");         // все значения для name="tags[]"
formData.has("email");           // boolean
formData.set("email", "new@x.y"); // замена
formData.append("tags", "js");   // добавление
formData.delete("tags");

// Файлы
const file = formData.get("avatar"); // File объект
file.name;        // "photo.jpg"
file.type;        // "image/jpeg"
file.size;        // байты
file.lastModified; // timestamp
const text = await file.text();    // как текст
const arrayBuffer = await file.arrayBuffer();
const stream = file.stream();      // ReadableStream
```

### Валидация (HTML5 + JS)
```html
<input name="email" type="email" required pattern="[^@]+@[^@]+\.[^@]+" title="Неверный email">
<input name="age" type="number" min="18" max="100" required>
<textarea name="bio" maxlength="500" required></textarea>
```

```javascript
const form = document.querySelector("form");
const email = form.elements.email;

// Встроенная валидация
email.validity.valid;           // boolean
email.validity.valueMissing;    // required не заполнено
email.validity.typeMismatch;    // type="email" но не email
email.validity.patternMismatch; // не проходит pattern
email.validity.tooLong;         // превышен maxlength
email.validity.rangeUnderflow;  // меньше min
email.validity.rangeOverflow;   // больше max

email.validationMessage;        // текст ошибки браузера
email.checkValidity();          // true/false, показывает UI ошибку
email.reportValidity();         // true/false, фокусирует и показывает UI

// Кастомная валидация
email.setCustomValidity("");    // сброс
if (customCheck(email.value)) {
  email.setCustomValidity("Этот email уже занят");
}
email.reportValidity();

// Отключение браузерной валидации
form.noValidate = true; // или <form novalidate>

// События
form.addEventListener("submit", (e) => {
  if (!form.checkValidity()) {
    e.preventDefault(); // не отправлять
    // показываем свои ошибки
  }
});

// Событие invalid (на элементе, не пузырится!)
email.addEventListener("invalid", (e) => {
  e.preventDefault(); // отменяем браузерный UI
  showCustomError(email, email.validationMessage);
});
```

### Отправка формы
```javascript
// Обычный submit (перезагрузка страницы)
<form action="/api/login" method="POST">...</form>

// AJAX отправка
form.addEventListener("submit", async (e) => {
  e.preventDefault();

  const formData = new FormData(form);
  const submitBtn = form.querySelector('[type="submit"]');
  submitBtn.disabled = true;

  try {
    const response = await fetch(form.action, {
      method: form.method,
      body: formData
    });

    if (response.ok) {
      const data = await response.json();
      showSuccess(data.message);
    } else {
      const error = await response.json();
      showErrors(error.fields); // { email: "Неверный формат" }
    }
  } catch (err) {
    showError("Сетевая ошибка");
  } finally {
    submitBtn.disabled = false;
  }
});

// Прогресс загрузки файлов
const xhr = new XMLHttpRequest(); // fetch не даёт progress!

  if (e.lengthComputable) {
    const percent = (e.loaded / e.total) * 100;
    progressBar.value = percent;
  }

xhr.send(formData);
```

## Типичные ошибки

| Ошибка | Правильно |
|--------|-----------|
| `el.innerHTML = userInput` — XSS | `el.textContent = userInput` или создавайте элементы через `createElement` |
| `querySelectorAll` возвращает массив | Это `NodeList` — используйте `[...els]` или `Array.from(els)` для методов массива |
| `addEventListener` без удаления — утечка памяти | Используйте `{ once: true }` или `AbortController` / `removeEventListener` в cleanup |
| `fetch` не бросает на 404/500 | Всегда проверяйте `response.ok` или `status` |
| `await response.json()` дважды | Тело можно прочитать один раз — используйте `response.clone()` |
| `localStorage` для больших данных (лимит ~5MB) | Используйте `IndexedDB` для больших объёмов |
| `document.cookie` читает HttpOnly куки | HttpOnly куки недоступны из JS — это защита от XSS |
| `history.pushState` не вызывает `popstate` | `popstate` срабатывает только на действия пользователя (back/forward) |
| `form.submit()` не вызывает событие `submit` | Используйте `form.requestSubmit()` для триггера валидации и события |
| `FormData` не отправляет пустые файлы | `<input type="file" required>` — браузер сам заблокирует отправку |
