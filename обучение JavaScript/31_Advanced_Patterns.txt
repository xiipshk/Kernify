```typescript
namespace OrderContext {
```

### 1.2 Tactical Design: Aggregates, Entities, Value Objects

```typescript
// Value Object — иммутабелен, сравнивается по значению
class Money {
  constructor(
    public readonly amount: number,
    public readonly currency: Currency = 'USD'
  ) {
    if (amount < 0) throw new Error('Negative amount');
    if (!Number.isInteger(amount * 100)) throw new Error('Precision > 2 decimals');
  }

  add(other: Money): Money {
    if (this.currency !== other.currency) throw new Error('Currency mismatch');
    return new Money(this.amount + other.amount, this.currency);
  }

  equals(other: Money): boolean {
    return this.amount === other.amount && this.currency === other.currency;
  }
}

// Entity — имеет идентичность, изменяется
class OrderItem {
  constructor(
    public readonly id: OrderItemId,
    public readonly productId: ProductId,
    public quantity: number,
    public readonly unitPrice: Money
  ) {}

  changeQuantity(qty: number) {
    if (qty <= 0) throw new Error('Quantity must be positive');
    this.quantity = qty;
  }

  getTotal(): Money {
    return new Money(this.unitPrice.amount * this.quantity, this.unitPrice.currency);
  }
}

// Aggregate Root — точка входа, гарантирует инварианты
class Order {
  private constructor(
    public readonly id: OrderId,
    public readonly customerId: CustomerId,
    private _items: OrderItem[] = [],
    private _status: OrderStatus = 'draft',
    private _domainEvents: DomainEvent[] = []
  ) {}

  static create(cmd: PlaceOrderCommand): Order {
    const order = new Order(
      OrderId.generate(),
      cmd.customerId,
      cmd.items.map(i => new OrderItem(...))
    );
    order.addDomainEvent(new OrderCreated(order.id));
    return order;
  }

  addItem(item: OrderItem) {
    if (this._status !== 'draft') throw new Error('Cannot modify confirmed order');
    this._items.push(item);
  }

  confirm() {
    if (this._items.length === 0) throw new Error('Empty order');
    this._status = 'confirmed';
    this.addDomainEvent(new OrderConfirmed(this.id));
  }

  get domainEvents(): DomainEvent[] {
    return [...this._domainEvents];
  }

  clearEvents() { this._domainEvents = []; }
}
```

### 1.3 CQRS (Command Query Responsibility Segregation)

```typescript
// Commands — изменяют состояние
interface Command { type: string; payload: any; metadata?: Record<string, any>; }

class CommandBus {
  private handlers = new Map<string, (cmd: Command) => Promise<void>>();

  register<T extends Command>(type: string, handler: (cmd: T) => Promise<void>) {
    this.handlers.set(type, handler as any);
  }

  async dispatch(cmd: Command) {
    const handler = this.handlers.get(cmd.type);
    if (!handler) throw new Error(`No handler for ${cmd.type}`);
    await handler(cmd);
  }
}

// Queries — только чтение, оптимизированы под чтение
interface Query<TResult> { type: string; payload: any; }

class QueryBus {
  private handlers = new Map<string, (query: Query<any>) => Promise<any>>();

  register<TResult>(type: string, handler: (q: Query<TResult>) => Promise<TResult>) {
    this.handlers.set(type, handler);
  }

  async ask<TResult>(query: Query<TResult>): Promise<TResult> {
    const handler = this.handlers.get(query.type);
    if (!handler) throw new Error(`No handler for ${query.type}`);
    return handler(query);
  }
}

// Read Models (Projections) — денормализованные представления
class OrderSummaryProjection {
  constructor(private readModel: OrderSummaryReadModel) {}

  @EventHandler(OrderConfirmed)
  async onOrderConfirmed(event: OrderConfirmed) {
    await this.readModel.upsert({
      orderId: event.orderId,
      customerId: event.customerId,
      total: event.total,
      status: 'confirmed',
      createdAt: event.timestamp
    });
  }
}
```

### 1.4 Event Sourcing

```typescript
// Event Store — append-only логи событий
interface EventStore {
  append(streamId: string, events: DomainEvent[], expectedVersion: number): Promise<void>;
  getStream(streamId: string): Promise<DomainEvent[]>;
  subscribe(streamId: string, handler: (event: DomainEvent) => void): () => void;
}

// Aggregate восстанавливается из событий
class Order {
  private _version = 0;

  static async reconstruct(eventStore: EventStore, orderId: OrderId): Promise<Order> {
    const events = await eventStore.getStream(`order-${orderId}`);
    const order = new Order(orderId, ...);
    events.forEach(e => order.apply(e));
    return order;
  }

  private apply(event: DomainEvent) {
    switch (event.type) {
      case 'OrderCreated': this.customerId = event.customerId; break;
      case 'ItemAdded': this._items.push(reconstructItem(event.item)); break;
      case 'OrderConfirmed': this._status = 'confirmed'; break;
    }
    this._version++;
  }

  // Новые события накапливаются для сохранения
  private _pendingEvents: DomainEvent[] = [];

  confirm() {
    if (this._status !== 'draft') throw new Error('Already confirmed');
    this._status = 'confirmed';
    this._pendingEvents.push(new OrderConfirmed(this.id, this._version + 1));
  }

  get pendingEvents() { return this._pendingEvents; }
  markEventsCommitted() { this._pendingEvents = []; }
}
```

## 2. Метапрограммирование

### 2.1 Proxy — перехват операций

```typescript
// Реактивность (как в Vue 3)
function reactive<T extends object>(target: T): T {
  const handlers: ProxyHandler<T> = {
    get(target, prop, receiver) {
      track(target, prop); // Dependency tracking
      const value = Reflect.get(target, prop, receiver);
      return typeof value === 'object' && value !== null ? reactive(value) : value;
    },
    set(target, prop, value, receiver) {
      const oldValue = target[prop];
      const result = Reflect.set(target, prop, value, receiver);
      if (oldValue !== value) trigger(target, prop); // Notify subscribers
      return result;
    },
    deleteProperty(target, prop) {
      const had = prop in target;
      const result = Reflect.deleteProperty(target, prop);
      if (had) trigger(target, prop);
      return result;
    },
    ownKeys(target) {
      track(target, Symbol.iterator);
      return Reflect.ownKeys(target);
    }
  };
  return new Proxy(target, handlers);
}

// Валидация через Proxy
function validated<T extends object>(target: T, schema: Record<string, (v: any) => boolean>): T {
  return new Proxy(target, {
    set(obj, prop, value) {
      const validator = schema[prop as string];
      if (validator && !validator(value)) {
        throw new Error(`Invalid value for ${String(prop)}: ${value}`);
      }
      return Reflect.set(obj, prop, value);
    }
  });
}

const user = validated({ age: 25, email: 'a@b.c' }, {
  age: v => Number.isInteger(v) && v >= 0 && v <= 150,
  email: v => /^[^@]+@[^@]+\.[^@]+$/.test(v)
});
user.age = 30; // OK
user.age = -5; // Error!
```

### 2.2 Reflect — низкоуровневые операции

```typescript
// Reflect — безопасные аналоги операторов
Reflect.get(obj, 'prop', receiver);        // obj[prop]
Reflect.set(obj, 'prop', value, receiver); // obj[prop] = value
Reflect.has(obj, 'prop');                  // 'prop' in obj
Reflect.deleteProperty(obj, 'prop');       // delete obj.prop
Reflect.construct(Class, args, newTarget); // new Class(...args)
Reflect.apply(fn, thisArg, args);          // fn.apply(thisArg, args)
Reflect.defineProperty(obj, prop, desc);   // Object.defineProperty
Reflect.getOwnPropertyDescriptor(obj, prop);
Reflect.ownKeys(obj);                      // Object.getOwnPropertyNames + Symbols
Reflect.preventExtensions(obj);            // Object.preventExtensions
Reflect.isExtensible(obj);                 // Object.isExtensible

// Пример: Generic Wrapper
function traceMethodCalls<T>(obj: T): T {
  return new Proxy(obj, {
    get(target, prop, receiver) {
      const value = Reflect.get(target, prop, receiver);
      if (typeof value === 'function') {
        return function(...args: any[]) {
          console.log(`→ ${String(prop)}(${args.join(', ')})`);
          const result = Reflect.apply(value, target, args);
          console.log(`← ${String(prop)} =>`, result);
          return result;
        };
      }
      return value;
    }
  });
}
```

### 2.3 Decorators (Stage 3 / TypeScript)

```typescript
// Class decorator
function sealed(constructor: Function) {
  Object.seal(constructor.prototype);
  Object.seal(constructor);
}

@sealed
class UserService {}

// Method decorator
function logged(target: Object, key: string | Symbol, descriptor: PropertyDescriptor) {
  const original = descriptor.value;
  descriptor.value = function(...args: any[]) {
    console.log(`Calling ${String(key)} with`, args);
    return original.apply(this, args);
  };
}

class Calculator {
  @logged
  add(a: number, b: number) { return a + b; }
}

// Property decorator (TC39 Stage 3 — другой синтаксис)
function range(min: number, max: number) {
  return function(target: Object, propertyKey: string) {
    let value: number;
    Object.defineProperty(target, propertyKey, {
      get() { return value; },
      set(v: number) {
        if (v < min || v > max) throw new Error(`Value ${v} not in [${min}, ${max}]`);
        value = v;
      }
    });
  };
}

class Thermostat {
  @range(0, 100)
  celsius: number = 20;
}

// Parameter decorator
function validate(schema: any) {
  return function(target: Object, key: string, parameterIndex: number) {
    // Сохраняем метаданные для runtime проверки
  };
}

function createUser(@validate(userSchema) userData: UserData) {}
```

### 2.4 Symbols — уникальные ключи и well-known symbols

```typescript
// Приватные "символьные" свойства
const PRIVATE = Symbol('private');
const METADATA = Symbol('metadata');

class SecureStore {
  [PRIVATE] = new Map();

  set(key: string, value: any) { this[PRIVATE].set(key, value); }
  get(key: string) { return this[PRIVATE].get(key); }
}

// Well-known Symbols
class MyArray extends Array {
  // Переопределение Symbol.iterator
  *[Symbol.iterator]() {
    for (let i = this.length - 1; i >= 0; i--) yield this[i];
  }

  // Symbol.species — конструктор для производных методов (map, filter...)
  static get [Symbol.species]() { return Array; }

  // Symbol.toStringTag — Object.prototype.toString
  get [Symbol.toStringTag]() { return 'MyArray'; }

  // Symbol.hasInstance — instanceof
  static [Symbol.hasInstance](instance: any) {
    return Array.isArray(instance) && instance.every(Number.isInteger);
  }
}

const arr = new MyArray(1, 2, 3);
[...arr]; // [3, 2, 1]
arr.map(x => x * 2) instanceof MyArray; // false — вернёт обычный Array!
Object.prototype.toString.call(arr); // "[object MyArray]"
[1,2,3] instanceof MyArray; // true!
```

## 3. Компиляция и трансформация кода

### 3.1 Babel Plugin Development

```javascript
// babel-plugin-auto-bind.js
// Автоматически биндит методы класса в конструкторе

export default function ({ types: t }) {
  return {
    visitor: {
      ClassMethod(path) {
        if (path.node.kind !== 'method') return;
        if (path.node.static) return;
        if (path.node.decorators?.some(d => t.isIdentifier(d.expression, { name: 'nobind' }))) return;

        const className = path.findParent(p => p.isClassDeclaration())?.node.id?.name;
        if (!className) return;

        // Находим или создаём конструктор
        let constructor = path.getSibling(path.key).get('body').get('body').find(
          p => p.isClassMethod({ kind: 'constructor' })
        );

        if (!constructor) {
          // Создаём конструктор
          constructor = t.classMethod('constructor', t.identifier('constructor'), [], t.blockStatement([]));
          path.getSibling(path.key).get('body').unshiftContainer('body', constructor);
        }

        // Добавляем this.method = this.method.bind(this)
        const bindStmt = t.expressionStatement(
          t.assignmentExpression(
            '=',
            t.memberExpression(t.thisExpression(), t.identifier(path.node.key.name)),
            t.callExpression(
              t.memberExpression(
                t.memberExpression(t.thisExpression(), t.identifier(path.node.key.name)),
                t.identifier('bind')
              ),
              [t.thisExpression()]
            )
          )
        );
        constructor.get('body').pushContainer('body', bindStmt);
      }
    }
  };
}
```

### 3.2 TypeScript Compiler API

```typescript
// ts-transformer.ts — кастомный трансформер для TypeScript
import * as ts from 'typescript';

function createTransformer(program: ts.Program): ts.TransformerFactory<ts.SourceFile> {
  return (context) => (sourceFile) => {
    const visitor: ts.Visitor = (node) => {
      // Заменяем console.log в production на noop
      if (ts.isCallExpression(node) &&
          ts.isPropertyAccessExpression(node.expression) &&
          node.expression.expression.getText() === 'console' &&
          node.expression.name.getText() === 'log') {

        if (process.env.NODE_ENV === 'production') {
          return ts.createVoidZero();
        }
      }

      // Автоматическое добавление __filename / __dirname
      if (ts.isIdentifier(node) && (node.text === '__filename' || node.text === '__dirname')) {
        return ts.createStringLiteral(sourceFile.fileName);
      }

      return ts.visitEachChild(node, visitor, context);
    };
    return ts.visitNode(sourceFile, visitor);
  };
}

// Использование в программе компиляции
const program = ts.createProgram(['src/index.ts'], {
  target: ts.ScriptTarget.ES2022,
  module: ts.ModuleKind.ESNext
});

const emitResult = program.emit(undefined, undefined, undefined, undefined, {
  before: [createTransformer(program)]
});
```

## 5. Plugin Architecture / Microkernel

### 5.1 Extensible Core

```typescript
// core/types.ts
interface Plugin {
  name: string;
  version: string;
  init(core: CoreAPI): Promise<void> | void;
  dispose?(): Promise<void> | void;
}

interface CoreAPI {
  registerCommand(name: string, handler: CommandHandler): void;
  registerQuery(name: string, handler: QueryHandler): void;
  registerExtensionPoint(name: string, extension: any): void;
  getExtensionPoint(name: string): any;
  eventBus: EventBus;
  config: Config;
  logger: Logger;
}

// core/kernel.ts
class Kernel {
  private plugins = new Map<string, Plugin>();
  private api: CoreAPI;

  constructor(private config: Config) {
    this.api = this.createAPI();
  }

  async loadPlugin(plugin: Plugin) {
    if (this.plugins.has(plugin.name)) {
      throw new Error(`Plugin ${plugin.name} already loaded`);
    }
    await plugin.init(this.api);
    this.plugins.set(plugin.name, plugin);
  }

  async unloadPlugin(name: string) {
    const plugin = this.plugins.get(name);
    if (!plugin) return;
    await plugin.dispose?.();
    this.plugins.delete(name);
  }

  async start() {
    // Загрузка встроенных плагинов
    await this.loadPlugin(new AuthPlugin());
    await this.loadPlugin(new StoragePlugin());
    // Пользовательские плагины из конфига
    for (const pluginSpec of this.config.plugins) {
      const plugin = await import(pluginSpec.path);
      await this.loadPlugin(plugin.default);
    }
  }
}
```

## 6. Advanced TypeScript Patterns

### 6.1 Template Literal Types

```typescript
// Генерация типов из строк
type EventName<T extends string> = `on${Capitalize<T>}`;
// "click" → "onClick"

type HTTPMethod = 'GET' | 'POST' | 'PUT' | 'DELETE' | 'PATCH';
type Endpoint = `/api/${string}`;
type Route = `${HTTPMethod} ${Endpoint}`;
// "GET /api/users"

// Рекурсивные типы для глубоких путей
type Path<T> = T extends object
  ? { [K in keyof T]: K extends string
      ? T[K] extends object
        ? `${K}.${Path<T[K]>}`
        : K
      : never }[keyof T]
  : never;

type UserPaths = Path<{ user: { profile: { name: string } } }>;
// "user.profile.name"
```

### 6.2 Conditional Types & Infer

```typescript
// Извлечение типа из Promise
type UnwrapPromise<T> = T extends Promise<infer U> ? U : T;

// Тип возвращаемого значения функции
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never;

// Параметры функции
type Parameters<T> = T extends (...args: infer P) => any ? P : never;

// Constructor parameters
type ConstructorParameters<T> = T extends new (...args: infer P) => any ? P : never;

// Instance type
type InstanceType<T> = T extends new (...args: any[]) => infer R ? R : never;

// Distributive conditional types
type ToArray<T> = T extends any ? T[] : never;
type Result = ToArray<string | number>; // string[] | number[]

// Exclude/Extract
type NonFunctionKeys<T> = { [K in keyof T]: T[K] extends Function ? never : K }[keyof T];
```

4. **Babel Plugin**: Плагин, удаляющий `console.log` и `debugger` в production
5. **Codegen**: Настройте OpenAPI → TS client + GraphQL Codegen в проекте
6. **Plugin System**: Создайте ядро приложения с динамической загрузкой плагинов
7. **Types**: Используйте Branded Types для ID сущностей в проекте урока 21

## Типичные ошибки

| Ошибка | Решение |
|--------|---------|
| Proxy ломает `===` проверки | Используйте `WeakMap` для хранения оригиналов |
| Декораторы не работают без `experimentalDecorators` | Включите в tsconfig или используйте Stage 3 синтаксис |
| Event Sourcing — сложность snapshot'ов | Делайте снапшоты каждые N событий |
| CQRS — eventual consistency | Чётко документируйте задержки, используйте Optimistic UI |
| Кодогенерация — устаревшие типы | Регенерируйте в CI при изменении схемы |
