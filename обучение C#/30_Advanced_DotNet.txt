### Minimal APIs (.NET 6+)
```csharp

// Простой endpoint

.WithName("GetUser")
.Produces<User>(200)
.Produces(404);

// Группы и фильтры

    return Results.Created($"/api/v1/users/{id}", id);
})
.RequireAuthorization("AdminPolicy")
.AddEndpointFilter<ValidationFilter<CreateUserDto>>();

// Binding sources
app.MapGet("/search", ([AsParameters] SearchRequest request) => ...); // Query + Header + Route
```

### Endpoint Filters
```csharp
public class ValidationFilter<T> : IEndpointFilter
{
    public async ValueTask<object?> InvokeAsync(EndpointFilterInvocationContext context, EndpointFilterDelegate next)
    {
        var validator = context.HttpContext.RequestServices.GetService<IValidator<T>>();
        if (validator is null) return await next(context);

        var argument = context.Arguments.OfType<T>().FirstOrDefault();
        if (argument is not null)
        {
            var result = await validator.ValidateAsync(argument, context.HttpContext.RequestAborted);
            if (!result.IsValid)
                return Results.ValidationProblem(result.ToDictionary());
        }
        return await next(context);
    }
}

// Global filter
app.AddEndpointFilter(async (ctx, next) =>
{
    var logger = ctx.HttpContext.RequestServices.GetRequiredService<ILogger<Program>>();
    logger.LogInformation("Executing {Endpoint}", ctx.HttpContext.GetEndpoint()?.DisplayName);
    return await next(ctx);
});
```

### Interceptors
```csharp
public class AuditingInterceptor : SaveChangesInterceptor
{
    public override InterceptionResult<int> SavingChanges(
        DbContextEventData eventData, InterceptionResult<int> result)
    {
        var entries = eventData.Context!.ChangeTracker.Entries<IAuditable>();
        foreach (var entry in entries)
        {
            if (entry.State == EntityState.Added)
            {
                entry.Entity.CreatedAt = DateTimeOffset.UtcNow;
                entry.Entity.CreatedBy = _currentUserService.UserId;
            }
            else if (entry.State == EntityState.Modified)
            {
                entry.Entity.UpdatedAt = DateTimeOffset.UtcNow;
                entry.Entity.UpdatedBy = _currentUserService.UserId;
            }
        }
        return base.SavingChanges(eventData, result);
    }
}

// Регистрация
builder.Services.AddDbContext<AppDbContext>((sp, options) =>
{
    options.UseNpgsql(connectionString)
           .AddInterceptors(sp.GetRequiredService<AuditingInterceptor>());
});
```

### Query Splitting и AsSplitQuery
```csharp
// Single query (cartesian explosion risk)
var users = await _context.Users
    .Include(u => u.Orders)
    .ThenInclude(o => o.Items)
    .ToListAsync();

// Split query (multiple SQL queries, better for large collections)
var users = await _context.Users
    .Include(u => u.Orders)
    .ThenInclude(o => o.Items)
    .AsSplitQuery()
    .ToListAsync();

// Global config
options.UseNpgsql(connectionString, o => o.UseQuerySplittingBehavior(QuerySplittingBehavior.SplitQuery));
```

### Change Tracking Optimization

## 3. Networking — Sockets, HTTP/2, gRPC, SignalR

### Sockets / TCP
```csharp
// TCP Server
var listener = new TcpListener(IPAddress.Any, 5000);
listener.Start();

while (true)
{
    var client = await listener.AcceptTcpClientAsync();
    _ = HandleClientAsync(client); // Fire and forget
}

async Task HandleClientAsync(TcpClient client)
{
    using var stream = client.GetStream();
    var buffer = new byte[4096];
    int read;
    while ((read = await stream.ReadAsync(buffer)) > 0)
    {
        var message = Encoding.UTF8.GetString(buffer, 0, read);
        await stream.WriteAsync(Encoding.UTF8.GetBytes($"Echo: {message}"));
    }
}
```

```csharp
// Server
public class UserService : UserProtoService.UserProtoServiceBase
{
    public override async Task<UserResponse> GetUser(GetUserRequest request, ServerCallContext context)
    {
        var user = await _repo.GetAsync(Guid.Parse(request.Id), context.CancellationToken);

    }
```

## 4. Background Services

### IHostedService и BackgroundService
```csharp
// Базовый BackgroundService
public class OrderProcessingService : BackgroundService
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<OrderProcessingService> _logger;
    private readonly TimeSpan _interval = TimeSpan.FromMinutes(5);

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // Initial delay
        await Task.Delay(TimeSpan.FromSeconds(30), stoppingToken);

        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                using var scope = _scopeFactory.CreateScope();
                var processor = scope.ServiceProvider.GetRequiredService<IOrderProcessor>();
                await processor.ProcessPendingAsync(stoppingToken);
            }
            catch (Exception ex) when (!stoppingToken.IsCancellationRequested)
            {
                _logger.LogError(ex, "Order processing failed");
            }

            await Task.Delay(_interval, stoppingToken);
        }
    }
}

// Регистрация
builder.Services.AddHostedService<OrderProcessingService>();
```

### Periodic Tasks с Timer
```csharp
public class PeriodicHostedService : IHostedService, IDisposable
{
    private Timer? _timer;
    private readonly IServiceScopeFactory _scopeFactory;

    public Task StartAsync(CancellationToken ct)
    {
        _timer = new Timer(DoWork, null, TimeSpan.Zero, TimeSpan.FromHours(1));
        return Task.CompletedTask;
    }

    private async void DoWork(object? state)
    {
        using var scope = _scopeFactory.CreateScope();
        var service = scope.ServiceProvider.GetRequiredService<ICleanupService>();
        await service.CleanupAsync(CancellationToken.None);
    }

    public Task StopAsync(CancellationToken ct)
    {
        _timer?.Change(Timeout.Infinite, 0);
        return Task.CompletedTask;
    }

    public void Dispose() => _timer?.Dispose();
}
```

### Distributed Locks для масштабирования
```csharp
// Используя Redis (StackExchange.Redis)
public class DistributedLockService
{
    private readonly IDatabase _redis;

    public async Task<IDisposable?> TryAcquireLockAsync(string key, TimeSpan expiry)
    {
        var lockKey = $"lock:{key}";
        var token = Guid.NewGuid().ToString();

        var acquired = await _redis.StringSetAsync(lockKey, token, expiry, When.NotExists);
        if (!acquired) return null;

        return new RedisLock(_redis, lockKey, token);
    }

    private class RedisLock : IDisposable
    {
        private readonly IDatabase _redis;
        private readonly string _key;
        private readonly string _token;

        public RedisLock(IDatabase redis, string key, string token)
        {
            _redis = redis; _key = key; _token = token;
        }

        public void Dispose()
        {
            // Lua script для атомарного удаление только своего токена
            var script = @"if redis.call('get', KEYS[1]) == ARGV[1] then return redis.call('del', KEYS[1]) else return 0 end";
            _redis.ScriptEvaluate(script, new RedisKey[] { _key }, new RedisValue[] { _token });
        }
    }
}

// Использование
public class SingletonBackgroundService : BackgroundService
{
    private readonly DistributedLockService _locks;

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            using var lockHandle = await _locks.TryAcquireLockAsync("daily-report", TimeSpan.FromMinutes(10));
            if (lockHandle != null)
            {
                // Only one instance runs this
                await GenerateDailyReportAsync(ct);
            }
            await Task.Delay(TimeSpan.FromHours(24), ct);
        }
    }
}
```

### Structured Logging
```csharp
// Program.cs
builder.Logging.ClearProviders();

public class OrderService
{

    public async Task<Order> CreateAsync(CreateOrderDto dto)
    {
        using var scope = _logger.BeginScope(new Dictionary<string, object>
        {
            ["CustomerId"] = dto.CustomerId,
            ["OrderId"] = Guid.NewGuid()

        try
        {
            var order = await _repository.CreateAsync(dto);
            _logger.LogInformation("Order {OrderId} created successfully", order.Id);
            return order;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to create order for customer {CustomerId}", dto.CustomerId);
            throw;
        }
    }
}
```

### Custom Metrics
```csharp
public class OrderMetrics
{
    private readonly Counter<long> _ordersCreated;
    private readonly Histogram<double> _orderValue;
    private readonly UpDownCounter<long> _activeOrders;

    public OrderMetrics(IMeterFactory meterFactory)
    {
        var meter = meterFactory.Create("MyApp.Orders");
        _ordersCreated = meter.CreateCounter<long>("orders.created", description: "Total orders created");
        _orderValue = meter.CreateHistogram<double>("orders.value", unit: "USD", description: "Order value distribution");
        _activeOrders = meter.CreateUpDownCounter<long>("orders.active", description: "Currently active orders");
    }

    public void OrderCreated(decimal value)
    {
        _ordersCreated.Add(1);
        _orderValue.Record((double)value);
        _activeOrders.Add(1);
    }

    public void OrderCompleted() => _activeOrders.Add(-1);
}
```

## 7. Типичные ошибки

| Ошибка | Правильно |
|--------|-----------|
| `AddDbContext` без пула | `AddDbContextPool` для высокой нагрузки |
| `Include` для больших коллекций | `AsSplitQuery()` или explicit loading |
| `HttpClient` создают вручную | `IHttpClientFactory` / `AddHttpClient` |
| BackgroundService без `IServiceScopeFactory` | Всегда создавайте scope для scoped сервисов |
| Отсутствие correlation ID | Middleware для генерации/проброса `X-Correlation-ID` |
| Логи без структуры | `Serilog` / `Microsoft.Extensions.Logging` с JSON formatter |
