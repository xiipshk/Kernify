## 2. Domain Layer — Rich Domain Model

```csharp
// Domain/Entities/Order.cs
public class Order : AggregateRoot<Guid>, IAuditable
{
    private readonly List<OrderItem> _items = new();
    private readonly List<IDomainEvent> _domainEvents = new();

    public CustomerId CustomerId { get; private set; }
    public OrderStatus Status { get; private set; }
    public ShippingAddress ShippingAddress { get; private set; }
    public Money Total => _items.Sum(i => i.LineTotal);
    public IReadOnlyCollection<OrderItem> Items => _items.AsReadOnly();
    public IReadOnlyCollection<IDomainEvent> DomainEvents => _domainEvents.AsReadOnly();

    // EF Core
    private Order() { }

    public static Order Create(CustomerId customerId, ShippingAddress address, IEnumerable<OrderItemData> items)
    {
        var order = new Order
        {
            Id = Guid.NewGuid(),
            CustomerId = customerId,
            ShippingAddress = address,
            Status = OrderStatus.Pending,
            CreatedAt = DateTimeOffset.UtcNow
        };

        foreach (var item in items)
            order.AddItem(item.ProductId, item.Quantity, item.UnitPrice);

        order.RaiseEvent(new OrderCreatedEvent(order.Id, order.CustomerId, order.Total));
        return order;
    }

    public void AddItem(ProductId productId, int quantity, Money unitPrice)
    {
        if (quantity <= 0) throw new DomainException("Quantity must be positive");
        if (unitPrice.Amount <= 0) throw new DomainException("Price must be positive");

        var existing = _items.FirstOrDefault(i => i.ProductId == productId);
        if (existing != null)
            existing.IncreaseQuantity(quantity);
        else
            _items.Add(new OrderItem(productId, quantity, unitPrice));
    }

    public void Confirm()
    {
        if (Status != OrderStatus.Pending)
            throw new DomainException($"Cannot confirm order in {Status} status");
        if (!_items.Any())
            throw new DomainException("Order must have at least one item");

        Status = OrderStatus.Confirmed;
        RaiseEvent(new OrderConfirmedEvent(Id));
    }

    public void Ship(TrackingNumber trackingNumber)
    {
        if (Status != OrderStatus.Confirmed)
            throw new DomainException("Only confirmed orders can be shipped");

        Status = OrderStatus.Shipped;
        RaiseEvent(new OrderShippedEvent(Id, trackingNumber));
    }

    public void Cancel(string reason)
    {
        if (Status is OrderStatus.Shipped or OrderStatus.Delivered)
            throw new DomainException("Cannot cancel shipped/delivered order");

        Status = OrderStatus.Cancelled;
        RaiseEvent(new OrderCancelledEvent(Id, reason));
    }

    public void ClearDomainEvents() => _domainEvents.Clear();
}
```

## 3. Application Layer — CQRS with MediatR

```csharp
// Application/Commands/CreateOrderCommand.cs
public record CreateOrderCommand(
    Guid CustomerId,
    ShippingAddressDto ShippingAddress,
    List<OrderItemDto> Items
) : IRequest<Result<Guid>>;

// Application/Validators/CreateOrderValidator.cs
public class CreateOrderValidator : AbstractValidator<CreateOrderCommand>
{
    public CreateOrderValidator()
    {
        RuleFor(x => x.CustomerId).NotEmpty();
        RuleFor(x => x.ShippingAddress).NotNull().SetValidator(new ShippingAddressValidator());
        RuleFor(x => x.Items).NotEmpty().ForEach(i => i.SetValidator(new OrderItemValidator()));
    }
}

// Application/Handlers/CreateOrderHandler.cs
public class CreateOrderHandler : IRequestHandler<CreateOrderCommand, Result<Guid>>
{
    private readonly IOrderRepository _orders;
    private readonly IProductRepository _products;
    private readonly IUnitOfWork _unitOfWork;
    private readonly IDomainEventDispatcher _dispatcher;

    public CreateOrderHandler(IOrderRepository orders, IProductRepository products,
        IUnitOfWork unitOfWork, IDomainEventDispatcher dispatcher)
    {
        _orders = orders;
        _products = products;
        _unitOfWork = unitOfWork;
        _dispatcher = dispatcher;
    }

    public async Task<Result<Guid>> Handle(CreateOrderCommand request, CancellationToken ct)
    {
        // Validate products exist and get prices
        var productIds = request.Items.Select(i => i.ProductId).Distinct().ToList();
        var products = await _products.GetByIdsAsync(productIds, ct);
        if (products.Count != productIds.Count)
            return Result.Failure<Guid>(Error.NotFound("Product.NotFound", "One or more products not found"));

        // Build order items with current prices
        var items = request.Items.Select(i =>
        {
            var product = products.First(p => p.Id == i.ProductId);
            return new OrderItemData(product.Id, i.Quantity, product.Price);
        });

        var order = Order.Create(
            new CustomerId(request.CustomerId),
            request.ShippingAddress.ToDomain(),
            items);

        await _orders.AddAsync(order, ct);
        await _unitOfWork.SaveChangesAsync(ct);
        await _dispatcher.DispatchAsync(order.DomainEvents, ct);

        return Result.Success(order.Id);
    }
}
```

```csharp
// Infrastructure/Persistence/Configurations/OrderConfiguration.cs
public class OrderConfiguration : IEntityTypeConfiguration<Order>
{
    public void Configure(EntityTypeBuilder<Order> builder)
    {
        builder.ToTable("Orders");
        builder.HasKey(o => o.Id);

        builder.Property(o => o.Id).ValueGeneratedNever();
        builder.Property(o => o.Status).HasConversion<string>().HasMaxLength(20);
        builder.Property(o => o.CreatedAt).IsRequired();

        builder.OwnsOne(o => o.ShippingAddress, sa =>
        {
            sa.Property(a => a.Street).HasMaxLength(200);
            sa.Property(a => a.City).HasMaxLength(100);
            sa.Property(a => a.PostalCode).HasMaxLength(20);
            sa.Property(a => a.Country).HasMaxLength(2);
        });

        builder.HasMany(o => o.Items)
            .WithOne()
            .HasForeignKey("OrderId")
            .OnDelete(DeleteBehavior.Cascade);

        builder.Navigation(o => o.Items).UsePropertyAccessMode(PropertyAccessMode.Field);
    }
}

// Infrastructure/Persistence/Interceptors/AuditingInterceptor.cs
public class AuditingInterceptor : SaveChangesInterceptor
{
    private readonly ICurrentUserService _currentUser;

    public AuditingInterceptor(ICurrentUserService currentUser) => _currentUser = currentUser;

    public override InterceptionResult<int> SavingChanges(
        DbContextEventData eventData, InterceptionResult<int> result)
    {
        UpdateAuditableEntities(eventData.Context);
        return base.SavingChanges(eventData, result);
    }

    private void UpdateAuditableEntities(DbContext? context)
    {
        if (context == null) return;

        var userId = _currentUser.UserId ?? "system";
        var now = DateTimeOffset.UtcNow;

        foreach (var entry in context.ChangeTracker.Entries<IAuditable>())
        {
            if (entry.State == EntityState.Added)
            {
                entry.Entity.CreatedAt = now;
                entry.Entity.CreatedBy = userId;
            }
            else if (entry.State == EntityState.Modified)
            {
                entry.Entity.UpdatedAt = now;
                entry.Entity.UpdatedBy = userId;
            }
        }
    }
}
```

## 5. API Layer — Minimal APIs with OpenAPI

```csharp
// Api/Endpoints/OrdersEndpoints.cs
public static class OrdersEndpoints
{
    public static void MapOrdersEndpoints(this IEndpointRouteBuilder app)
    {
        var group = app.MapGroup("/api/v1/orders")
            .RequireAuthorization()
            .WithTags("Orders")
            .WithOpenApi();

        group.MapGet("/", async (IMediator mediator, [AsParameters] GetOrdersQuery query, CancellationToken ct) =>
        {
            var result = await mediator.Send(query, ct);
            return result.IsSuccess ? Results.Ok(result.Value) : result.ToProblemDetails();
        })
        .Produces<PagedResult<OrderDto>>(200)
        .Produces(400);

        group.MapGet("/{id:guid}", async (Guid id, IMediator mediator, CancellationToken ct) =>
        {
            var result = await mediator.Send(new GetOrderQuery(id), ct);
            return result.IsSuccess ? Results.Ok(result.Value) : result.ToProblemDetails();
        })
        .Produces<OrderDto>(200)
        .Produces(404);

        group.MapPost("/", async (CreateOrderCommand command, IMediator mediator, CancellationToken ct) =>
        {
            var result = await mediator.Send(command, ct);
            return result.IsSuccess
                ? Results.Created($"/api/v1/orders/{result.Value}", result.Value)
                : result.ToProblemDetails();
        })
        .Produces<Guid>(201)
        .Produces(400)
        .AddEndpointFilter<ValidationFilter<CreateOrderCommand>>();

        group.MapPost("/{id:guid}/confirm", async (Guid id, IMediator mediator, CancellationToken ct) =>
        {
            var result = await mediator.Send(new ConfirmOrderCommand(id), ct);
            return result.IsSuccess ? Results.Ok() : result.ToProblemDetails();
        })
        .Produces(200)
        .Produces(400)
        .Produces(404);
    }
}

// Api/Extensions/ResultExtensions.cs
public static class ResultExtensions
{
    public static IResult ToProblemDetails(this Result result)
    {
        return result.Error?.Code switch
        {
            "Validation" => Results.ValidationProblem(result.Error.ToDictionary()),
            "NotFound" => Results.NotFound(result.Error.Message),
            "Conflict" => Results.Conflict(result.Error.Message),
            _ => Results.Problem(result.Error?.Message, statusCode: 500)
        };
    }
}
```

```csharp
// tests/MyProject.ArchitectureTests/ArchitectureTests.cs
[TestFixture]
public class ArchitectureTests
{
    private static readonly Assembly DomainAssembly = typeof(Order).Assembly;
    private static readonly Assembly ApplicationAssembly = typeof(CreateOrderHandler).Assembly;
    private static readonly Assembly InfrastructureAssembly = typeof(OrderRepository).Assembly;
    private static readonly Assembly ApiAssembly = typeof(Program).Assembly;

    [Test]
    public void Domain_ShouldNotReferenceOtherLayers()
    {
        var types = Types.InAssembly(DomainAssembly);
        types.ShouldNot()
            .HaveDependencyOn(ApplicationAssembly.GetName().Name!)
            .And.ShouldNot()
            .HaveDependencyOn(InfrastructureAssembly.GetName().Name!)
            .And.ShouldNot()
            .HaveDependencyOn(ApiAssembly.GetName().Name!);
    }

    [Test]
    public void Application_ShouldNotReferenceInfrastructureOrApi()
    {
        Types.InAssembly(ApplicationAssembly)
            .ShouldNot()
            .HaveDependencyOn(InfrastructureAssembly.GetName().Name!)
            .And.ShouldNot()
            .HaveDependencyOn(ApiAssembly.GetName().Name!);
    }

    [Test]
    public void Handlers_ShouldBeInApplicationLayer()
    {
        Types.InAssembly(ApplicationAssembly)
            .That().ImplementInterface(typeof(IRequestHandler<,>))
            .Should().ResideInNamespace("MyProject.Application");
    }

    [Test]
    public void Entities_ShouldBeSealedOrHaveProtectedConstructor()
    {
        Types.InAssembly(DomainAssembly)
            .That().Inherit(typeof(AggregateRoot<>))
            .Should().HaveProtectedConstructor();
    }
}
```

```yaml
    runs-on: ubuntu-latest

      - uses: actions/checkout@v4

  performance-test:
    runs-on: ubuntu-latest
    needs: build-and-test
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with: { dotnet-version: ${{ env.DOTNET_VERSION }} }
      - name: Run Benchmarks
        run: dotnet run -c Release --project tests/MyProject.PerformanceTests -- --filter **

  docker-build:
    runs-on: ubuntu-latest
    needs: [build-and-test, security-scan]
    if: github.event_name == 'push'
    steps:
      - uses: actions/checkout@v4
      - name: Build API Image
        run: docker build -f docker/Dockerfile.api -t ${{ env.DOCKER_REGISTRY }}/${{ github.repository }}/api:${{ github.sha }} .
      - name: Build Worker Image
        run: docker build -f docker/Dockerfile.worker -t ${{ env.DOCKER_REGISTRY }}/${{ github.repository }}/worker:${{ github.sha }} .
      - name: Trivy Image Scan
        run: |
          trivy image --severity HIGH,CRITICAL ${{ env.DOCKER_REGISTRY }}/${{ github.repository }}/api:${{ github.sha }}
          trivy image --severity HIGH,CRITICAL ${{ env.DOCKER_REGISTRY }}/${{ github.repository }}/worker:${{ github.sha }}
      - name: Push
        run: |
          echo ${{ secrets.GITHUB_TOKEN }} | docker login ${{ env.DOCKER_REGISTRY }} -u ${{ github.actor }} --password-stdin
          docker push ${{ env.DOCKER_REGISTRY }}/${{ github.repository }}/api:${{ github.sha }}
          docker push ${{ env.DOCKER_REGISTRY }}/${{ github.repository }}/worker:${{ github.sha }}

  deploy-staging:
    runs-on: ubuntu-latest
    needs: docker-build
    if: github.ref == 'refs/heads/develop'
    steps:
      - name: Deploy to Staging
        run: helm upgrade --install myproject-staging helm/myproject -n staging --set image.tag=${{ github.sha }}

  deploy-production:
    runs-on: ubuntu-latest
    needs: docker-build

    environment: production
    steps:
      - name: Deploy to Production
        run: helm upgrade --install myproject-prod helm/myproject -n production --set image.tag=${{ github.sha }}
```

```markdown
# 1. Use Clean Architecture

**Date**: 2024-01-15
**Status**: Accepted

## Context
Need to decide on architectural pattern for new order management system.

## Decision
Adopt Clean Architecture with Domain, Application, Infrastructure, API layers.

## Consequences
- **Positive**: Testability, separation of concerns, framework independence
- **Negative**: More boilerplate, steeper learning curve
- **Mitigation**: Use templates, code generators, team training
```

```markdown
# docs/runbooks/order-processing-failure.md

# Runbook: Order Processing Failure

## Symptoms
- Orders stuck in "Pending" status
- Increased `orders.failed` metric
- Alert: OrderProcessingFailureRate > 5%

## Diagnosis
1. Check logs: `kubectl logs -l app=myproject-worker -n production`
2. Check queue depth: `redis-cli LLEN orders:pending`
3. Check database locks: `SELECT * FROM pg_locks WHERE relation = 'orders'`

## Resolution
1. **Transient DB error**: Restart worker pods
2. **Payment gateway down**: Enable fallback mode, alert on-call
3. **Deadlock**: Kill blocking queries, investigate indexes

## Prevention
- Add circuit breaker for payment gateway
- Implement idempotency keys
- Add distributed lock for order processing
```

```csharp
public class CapacityMetrics : BackgroundService
{
    private readonly IMetricsCollector _metrics;
    private readonly ILogger<CapacityMetrics> _logger;

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            var cpu = _metrics.GetCpuUsage();
            var memory = _metrics.GetMemoryUsage();
            var threadPool = _metrics.GetThreadPoolStats();
            var dbConnections = _metrics.GetDbConnectionPoolUsage();

            _logger.LogInformation(
                "Capacity: CPU={Cpu}% Mem={Memory}MB Threads={Threads} DB={Db}%",
                cpu, memory, threadPool.AvailableThreads, dbConnections.UsagePercent);

            if (cpu > 80 || memory > 85 || dbConnections.UsagePercent > 90)
            {
                _logger.LogWarning("Capacity threshold exceeded!");
                // Alert via webhook
            }

            await Task.Delay(TimeSpan.FromMinutes(5), ct);
        }
    }
}
```
