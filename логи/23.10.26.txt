  .mode column               -- красивый вывод

  .quit                      -- выход

  .headers on                -- заголовки

  .quit                      -- выход

  команда)

  # Все пользователи

  c=sqlite3.connect('C:/bot_project/bot.db');

  users').fetchall())"

  # Все языки пользователей
